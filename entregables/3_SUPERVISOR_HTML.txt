'use strict';

/* =====================================================
   UGM 2026 · PLATAFORMA DE SUPERVISIÓN

const V={
 pagina:'control',
 fecha:'',
 datos:null,
 consultando:false,
 timer:null,
 filtroSol:'sinverificar',
 colaIds:null,
 detFiltro:'todos',
 feedVistos:new Set(),
 firmas:{}
};

// Compatibilidad con la carga de base y comprobantes.
const S={
 get token(){return UGM.token;},
 cargandoPacientes:false,
 archivo:null,
 previewURL:null
};

function puedeGestionar() {
 return UGM.perfil==='SUPERVISOR';
}

const TITULOS={
 control:['Centro de control','Operación en tiempo real'],
 solicitudes:['Solicitudes','Revisión y aprobación de sobrecupos'],
 asistencia:['Asistencia','Reporte de asistieron y no asistieron'],
 personal:['Personal en línea','Sesiones y actividad del equipo'],
 cupos:['Cupos y límites','Disponibilidad médica por especialidad'],
 basepacientes:['Base de pacientes','Base enviada por Reportería'],
 actualizacion:['Agenda del día','Reporte de Citas del sistema hospitalario'],
 auditoria:['Auditoría','Historial de operaciones']
};

function page(n) {
 if(!TITULOS[n])return;

 const cambio=V.pagina!==n;
 const antes=V.pagina;
 V.pagina=n;

 mostrarPagina(n,TITULOS[n][0],!cambio);
 $('subtitle').textContent=TITULOS[n][1];

 if(cambio) {
  V.firmas={};
  if(n==='asistencia'&&antes!=='asistencia') {
   refrescar(true);
  } else {
   dibujar(true);
  }
 }
}

/* =====================================================
   SINCRONIZACIÓN
   ===================================================== */

async function refrescar(forzar) {
 if(!UGM.token||UGM.importando)return;
 if(V.consultando&&!forzar)return;

 V.consultando=true;

 const gen=UGM.generacion;
 const fecha=V.fecha;

 try {
  const d=await api('tableroSupervisor',UGM.token,fecha,{
   detalle:V.pagina==='asistencia',
   limites:V.pagina==='cupos',
   actividad:'Supervisión · '+TITULOS[V.pagina][0]
  });

  if(gen!==UGM.generacion||fecha!==V.fecha)return;

  V.datos=d;
  dibujar(false);

  $('sync').textContent='● Sincronizado';
  $('sync').style.color='var(--green)';
  $('status').textContent='Última sincronización: '+
   new Date().toLocaleTimeString('es-SV');

 }catch(e) {

 }
}

// Resumen por especialidad: el total sumado de todas las
// especialidades (p. ej. 202) no sirve al supervisor,
// porque cada especialidad tiene su propio tope.
function totalesCupos() {
 return (V.datos.cupos||[]).reduce((t,r)=>{
  if(!r.habilitada)return t;
  t.habilitadas++;
  t.ocupados+=Number(r.ocupados)||0;
  if(r.disponibles>0)t.conCupo++;
  else if(r.urgenciaLibre>0)t.soloUrgencia++;
  else t.completas++;
  t.urgencias+=Number(r.urgenciaUsada)||0;
  return t;
 },{habilitadas:0,conCupo:0,soloUrgencia:0,completas:0,
   ocupados:0,urgencias:0});
}

function donut(cont,segmentos,centro,etiqueta) {
 const total=segmentos.reduce((t,s)=>t+s.valor,0);
 const r=15.915;

 let acumulado=0;

 const circulos=segmentos.map(s=>{
  const pct=total?s.valor/total*100:0;
  const html=`<circle r="${r}" cx="21" cy="21"
   stroke="${s.color}" stroke-dasharray="0 100"
   stroke-dashoffset="${-acumulado}" data-p="${pct}"></circle>`;
  acumulado+=pct;
  return html;
 }).join('');

 cont.innerHTML=`
  <div class="donut">
   <svg viewBox="0 0 42 42">
    <circle class="base" r="${r}" cx="21" cy="21"></circle>
    ${circulos}
   </svg>
   <div class="centro"><div>
    <strong data-n="${centro}">0</strong>
    <small>${esc(etiqueta)}</small>
   </div></div>
  </div>
  <div class="legend">
   ${segmentos.map(s=>`<span><i style="background:${s.color}"></i>
    ${esc(s.etiqueta)}<b>${s.valor}</b></span>`).join('')}
  </div>`;

 MOV.alVer(cont,()=>{
  cont.querySelectorAll('circle[data-p]').forEach((c,i)=>{
   const p=Number(c.dataset.p);
   if(!MOV.ok) {
    c.setAttribute('stroke-dasharray',p+' '+(100-p));
    return;
   }
   gsap.to(c,{attr:{'stroke-dasharray':p+' '+(100-p)},
    duration:1.1,ease:'power3.out',delay:i*.12});
  });

  const n=cont.querySelector('strong[data-n]');
  MOV.numero(n,Number(centro)||0);

  if(MOV.ok)gsap.fromTo(cont.querySelectorAll('.legend span'),
   {x:-10,opacity:0},{x:0,opacity:1,stagger:.06,duration:.35,
    clearProps:'transform,opacity'});
 });
}

async function gestionar(id,forzado) {
 const boton=$('btnSaveState')||$('btnLiberar');
 const estado=forzado||$('newState').value;
 const r=(V.datos.solicitudes||[]).find(x=>x.ID===id);

 if(['RECHAZADO','ANULADO'].includes(estado)) {
  if(forzado&&!$('obs').value.trim()) {
   aviso('Indica el motivo para liberar el cupo.');
   return;
  }
  const ok=await confirmar({
   titulo:estado==='RECHAZADO'?'RECHAZAR SOLICITUD':'ANULAR Y LIBERAR CUPO',
   html:`<p>Esto liberará <b>1 cupo</b> de
    <b>${esc(r.Especialidad)}</b> para el</p>
    <div class="dia-grande">${esc(fechaLarga(r.FechaSobrecupo))}</div>
    <p>Paciente: <b>${esc(r.Nombres)}</b></p>`,
   ok:estado==='RECHAZADO'?'Rechazar':'Anular y liberar',
   cancelar:'Volver'
  });
  if(!ok)return;
 }

 boton.disabled=true;
 boton.textContent='Guardando...';

 try {
  const tiene=r&&(r.TicketFrente||r.ComprobanteID);

  if(estado==='REGISTRADO'&&!S.archivo&&!tiene) {
   throw Error('Adjunta un comprobante antes de registrar.');
  }

  if(S.archivo) {
   const f=await prepararArchivo(S.archivo);
   const ok=await api('adjuntarComprobante',UGM.token,id,
    f.nombre,f.mime,f.base64);
   if(!ok||!ok.ok)throw Error('No se confirmó el comprobante.');
  }

  const res=await api('cambiarEstado',UGM.token,id,estado,
   $('obs').value);

  if(!res||!res.ok)throw Error('No se confirmó el cambio.');

  cerrarModal();
  MOV.exito();
  aviso('Solicitud '+id+': '+textoEstado(estado)+
   (['RECHAZADO','ANULADO'].includes(estado)?' · cupo liberado.':'.'));
  V.firmas={};
  await refrescar(true);

 }catch(e) {
  if(sesionTerminada(e,limpiarTodo))return;
  aviso(e.message);
  boton.disabled=false;
  boton.textContent=forzado?'Liberar cupo':'Guardar gestión';
 }
}

  nombre:'Pacientes agendados',

  detalle:'Pacientes que ya estaban agendados para la fecha (base de Reportería y agenda del sistema). El "Aumento 10%" se indica en la columna Tipo.'
 },
 SOBRECUPO:{
  nombre:'Pacientes adicionales (sobrecupo)',
  corto:'Adicional (sobrecupo)',
  hoja:'Adicionales sobrecupo',
  detalle:'Pacientes adicionales agendados como sobrecupo por los técnicos en la plataforma.'
 }
};

// Tipo de cita para pantallas y reportes.
function tipoCita(c) {
 if(grupoReporte(c)==='SOBRECUPO')return 'Sobrecupo';
 if(c.aumento)return 'Aumento 10%';
 if(c.origen==='REPORTERIA')return 'Reportería';
 return 'Agenda del sistema';
}

// Un sobrecupo que luego aparece en el reporte del
// sistema sigue contando como sobrecupo.
function grupoReporte(c) {
 return c.origen==='SOBRECUPO'||c.sobrecupo?'SOBRECUPO':'AGENDA';
}

function conteo(lista) {
 const t={citas:lista.length,asistio:0,noAsistio:0,sinMarcar:0};
 lista.forEach(c=>{
  if(c.asistencia==='ASISTIO')t.asistio++;
  else if(c.asistencia==='NO_ASISTIO')t.noAsistio++;
  else t.sinMarcar++;
 });
 const m=t.asistio+t.noAsistio;
 t.pct=m?Math.round(t.asistio/m*100):0;
 t.texto=m?t.pct+'%':'—';
 return t;
}

function porEspecialidad(lista) {
 const g={};
 lista.forEach(c=>{(g[c.especialidad]=g[c.especialidad]||[]).push(c);});
 return Object.keys(g)
  .map(e=>Object.assign({especialidad:e},conteo(g[e])))
  .sort((a,b)=>b.citas-a.citas);
}

function pintarOrigen() {
 const lista=V.datos.consultas||[];
 const filas=[
  ['Agendados',conteo(lista.filter(c=>grupoReporte(c)==='AGENDA'))],
  ['   · de ellos, aumento 10%',conteo(lista.filter(c=>grupoReporte(c)==='AGENDA'&&c.aumento))],
  ['Adicionales (sobrecupo)',conteo(lista.filter(c=>grupoReporte(c)==='SOBRECUPO'))],
  ['TOTAL',conteo(lista)]
 ];

 if(!cambio('origen',filas))return;

 $('origenRows').innerHTML=filas.map(([n,t])=>`
  <tr${n==='TOTAL'?' style="font-weight:800"':''}>
   <td><b>${n}</b></td>
   <td>${t.citas}</td>
   <td style="color:var(--green)">${t.asistio}</td>
   <td style="color:var(--red)">${t.noAsistio}</td>
   <td style="color:var(--yellow)">${t.sinMarcar}</td>
   <td>${t.texto}</td>
  </tr>`).join('');
}

function detalleFiltrado() {
 const esp=$('detEsp').value;
 const q=normTexto($('detBuscar').value);

 return (V.datos.consultas||[]).filter(c=>
  (!esp||c.especialidad===esp)&&
  (V.detFiltro==='todos'||c.asistencia===V.detFiltro)&&
  (!q||normTexto(c.paciente+' '+c.documento).includes(q))
 );
}

function textoAsistencia(v) {
 return v==='ASISTIO'?'Asistió':v==='NO_ASISTIO'?'No asistió':'Sin marcar';
}

 const filas=detalleFiltrado();

 if(!cambio('det',[$('detEsp').value,$('detBuscar').value,V.detFiltro,
   filas.map(c=>c.clave+c.asistencia)])&&!animar)return;

    c.asistencia==='NO_ASISTIO'?'var(--red)':'var(--yellow)'}">

const COLOR_REP={
 si:'#21A67A',no:'#D24B5D',sin:'#E0A526',
 azul:'#10456B',cian:'#0A9FD0',texto:'#12354E',gris:'#58768A'
};

function lienzo(w,h) {
 const c=document.createElement('canvas');
 c.width=w*2;c.height=h*2;
 const x=c.getContext('2d');
 x.scale(2,2);
 x.fillStyle='#FFFFFF';
 x.fillRect(0,0,w,h);
 x.font='14px Segoe UI, Arial, sans-serif';
 return {c,x};
}

// Dona: asistieron / no asistieron / sin marcar.
function graficaDona(t,titulo) {
 const W=560,H=300;
 const {c,x}=lienzo(W,H);

 x.fillStyle=COLOR_REP.texto;
 x.font='bold 17px Segoe UI, Arial, sans-serif';
 x.fillText(titulo,20,30);

 const seg=[
  ['Asistieron',t.asistio,COLOR_REP.si],
  ['No asistieron',t.noAsistio,COLOR_REP.no],
  ['Sin marcar',t.sinMarcar,COLOR_REP.sin]
 ];

 const cx=140,cy=170,r=95,g=34;
 let a=-Math.PI/2;

 if(!t.citas) {
  x.strokeStyle='#E3EDF3';x.lineWidth=g;
  x.beginPath();x.arc(cx,cy,r,0,Math.PI*2);x.stroke();
 }

 seg.forEach(([,v,col])=>{
  if(!v)return;
  const b=a+v/t.citas*Math.PI*2;
  x.strokeStyle=col;x.lineWidth=g;
  x.beginPath();x.arc(cx,cy,r,a,b);x.stroke();
  a=b;
 });

 x.textAlign='center';
 x.fillStyle=COLOR_REP.texto;
 x.font='bold 30px Segoe UI, Arial, sans-serif';
 x.fillText(t.texto,cx,cy+6);
 x.font='12px Segoe UI, Arial, sans-serif';
 x.fillStyle=COLOR_REP.gris;
 x.fillText('asistencia',cx,cy+26);
 x.textAlign='left';

 seg.forEach(([n,v,col],i)=>{
  const y=110+i*44;
  x.fillStyle=col;
  x.fillRect(290,y-12,16,16);
  x.fillStyle=COLOR_REP.texto;
  x.font='bold 18px Segoe UI, Arial, sans-serif';
  x.fillText(String(v),316,y+3);
  x.font='14px Segoe UI, Arial, sans-serif';
  x.fillStyle=COLOR_REP.gris;
  x.fillText(n+(t.citas?' · '+Math.round(v/t.citas*100)+'%':''),356,y+2);
 });

 x.fillStyle=COLOR_REP.gris;
 x.font='13px Segoe UI, Arial, sans-serif';
 x.fillText('Total de citas: '+t.citas,290,250);

 return c.toDataURL('image/png');
}

// Barras apiladas por especialidad (o por grupo).
function graficaBarras(filas,titulo) {
 const W=760,fila=30,H=90+Math.max(1,filas.length)*fila;
 const {c,x}=lienzo(W,H);

 x.fillStyle=COLOR_REP.texto;
 x.font='bold 17px Segoe UI, Arial, sans-serif';
 x.fillText(titulo,20,30);

 const ley=[['Asistieron',COLOR_REP.si],['No asistieron',COLOR_REP.no],
  ['Sin marcar',COLOR_REP.sin]];
 ley.forEach(([n,col],i)=>{
  x.fillStyle=col;x.fillRect(20+i*150,44,12,12);
  x.fillStyle=COLOR_REP.gris;x.font='12px Segoe UI, Arial, sans-serif';
  x.fillText(n,38+i*150,55);
 });

 const max=Math.max(1,...filas.map(f=>f.citas));
 const x0=250,ancho=W-x0-120;

 filas.forEach((f,i)=>{
  const y=72+i*fila;
  x.fillStyle=COLOR_REP.texto;
  x.font='13px Segoe UI, Arial, sans-serif';
  const et=f.especialidad.length>34?f.especialidad.slice(0,33)+'…':f.especialidad;
  x.fillText(et,20,y+15);

  let px=x0;
  [[f.asistio,COLOR_REP.si],[f.noAsistio,COLOR_REP.no],
   [f.sinMarcar,COLOR_REP.sin]].forEach(([v,col])=>{
   const w=v/max*ancho;
   x.fillStyle=col;
   x.fillRect(px,y+2,w,18);
   px+=w;
  });

  x.fillStyle=COLOR_REP.gris;
  x.fillText(f.asistio+'/'+f.citas+(f.asistio+f.noAsistio?' · '+f.texto:''),
   px+8,y+16);
 });

 return {url:c.toDataURL('image/png'),alto:H};
}

// Impresión: horizontal, ajustada al ancho de una página.
function paraImprimir(ws) {
 ws.pageSetup={paperSize:9,orientation:'landscape',fitToPage:true,
  fitToWidth:1,fitToHeight:0,
  margins:{left:.4,right:.4,top:.5,bottom:.5,header:.2,footer:.2}};
}

function estiloTitulo(celda,tamano) {
 celda.font={bold:true,size:tamano||16,color:{argb:'FFFFFFFF'}};
 celda.fill={type:'pattern',pattern:'solid',fgColor:{argb:'FF10456B'}};
 celda.alignment={vertical:'middle'};
}

function encabezados(fila) {
 fila.eachCell(c=>{
  c.font={bold:true,color:{argb:'FFFFFFFF'}};
  c.fill={type:'pattern',pattern:'solid',fgColor:{argb:'FF0A9FD0'}};
  c.alignment={vertical:'middle',horizontal:'center',wrapText:true};
  c.border={bottom:{style:'thin',color:{argb:'FF10456B'}}};
 });
 fila.height=22;
}

   i===4?ARGB.sin:'FF12354E'}};
  c.alignment={horizontal:'center',vertical:'middle'};
  c.fill={type:'pattern',pattern:'solid',fgColor:{argb:'FFF1F8FC'}};
 });
}

  String(a.horaCita||'').localeCompare(String(b.horaCita||''))||
  a.paciente.localeCompare(b.paciente)
 ).forEach((c,i)=>{
  const r=ws.getRow(inicio+2+i);
  r.values=[c.especialidad,c.paciente,c.documento,
   textoAsistencia(c.asistencia),c.marcadoPor||'',
   c.hora?horaCorta(c.hora):'',
   c.sobrecupo?textoEstado(c.sobrecupo):(c.estadoCita||''),
   tipoCita(c),c.horaCita||''];
  if(c.aumento&&grupoReporte(c)==='AGENDA') {
   r.getCell(8).font={bold:true,color:{argb:'FFB7791F'}};
  }
  const a=r.getCell(4);
  a.font={bold:true,color:{argb:c.asistencia==='ASISTIO'?ARGB.si:
   c.asistencia==='NO_ASISTIO'?ARGB.no:ARGB.sin}};
  if(i%2)r.eachCell(x=>{x.fill={type:'pattern',pattern:'solid',
   fgColor:{argb:'FFF6FAFD'}};});
 });

 ws.autoFilter={from:{row:inicio+1,column:1},
  to:{row:inicio+1,column:9}};
}

async function descargarReporte() {
 const d=V.datos;

 if(!d||!d.consultas) {
  aviso('Espera a que cargue el detalle de asistencia.');
  return;
 }

 const boton=$('btnReporte');
 boton.disabled=true;
 boton.innerHTML=icono('reloj')+' Generando reporte...';

 try {
  if(!window.ExcelJS)await cargarScript(EXCELJS_URL);
  if(!window.ExcelJS)throw Error('ExcelJS no disponible.');

  const lista=d.consultas;
  const wb=new ExcelJS.Workbook();
  wb.creator='UGM · Nuevo Hospital Rosales';
  wb.created=new Date();

  hojaResumen(wb,d,lista);
  hojaGrupo(wb,'AGENDA',lista.filter(c=>grupoReporte(c)==='AGENDA'),d);
  hojaGrupo(wb,'SOBRECUPO',lista.filter(c=>grupoReporte(c)==='SOBRECUPO'),d);

  const buffer=await wb.xlsx.writeBuffer();
  const blob=new Blob([buffer],{type:
   'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(blob);
  a.download='Reporte_Asistencia_'+d.fecha+'.xlsx';
  document.body.appendChild(a);
  a.click();
  setTimeout(()=>{URL.revokeObjectURL(a.href);a.remove();},3000);

  MOV.exito();
  aviso('Reporte descargado con resumen y gráficas.');

 }catch(e) {
  console.warn(e);
  aviso('No se pudieron generar las gráficas; se descarga el reporte simple.');
  descargarReporteSimple();

 }finally {
  boton.disabled=false;
  boton.innerHTML=icono('descargar')+' Descargar reporte Excel';
 }
}

function descargarReporteSimple() {
 const d=V.datos;

 if(!d||!d.consultas) {
  aviso('Espera a que cargue el detalle de asistencia.');
  return;
 }

 if(!window.XLSX) {
  aviso('No se pudo cargar el generador de Excel.');
  return;
 }

 const a=d.asistencia;
 const pct=(si,no)=>si+no?Math.round(si/(si+no)*100)+'%':'—';

 const resumen=[
  ['REPORTE DE ASISTENCIA · UGM · NUEVO HOSPITAL ROSALES'],
  ['Fecha de consulta',d.fecha],
  ['Generado',new Date().toLocaleString('es-SV')],
  ['Generado por',UGM.nombre],
  [],
  ['Especialidad','Citas','Asistieron','No asistieron',
   'Sin marcar','% asistencia'],
  ...a.especialidades.map(e=>[e.especialidad,e.citas,e.asistio,
   e.noAsistio,e.sinMarcar,pct(e.asistio,e.noAsistio)]),
  ['TOTAL',a.total.citas,a.total.asistio,a.total.noAsistio,
   a.total.sinMarcar,pct(a.total.asistio,a.total.noAsistio)]
 ];

 const fila=c=>[c.especialidad,c.paciente,c.documento,
  tipoCita(c),c.horaCita||'',
  textoAsistencia(c.asistencia),c.marcadoPor,
  c.hora?horaCorta(c.hora):''];

 const cab=['Especialidad','Paciente','Documento','Tipo','Hora cita',
  'Asistencia','Marcado por','Hora marca'];

 const hoja=(filas,anchos)=>{
  const h=XLSX.utils.aoa_to_sheet(filas);
  h['!cols']=anchos.map(w=>({wch:w}));
  return h;
 };

 const anchos=[30,38,16,20,10,14,26,10];
 const wb=XLSX.utils.book_new();

 XLSX.utils.book_append_sheet(wb,
  hoja(resumen,[34,10,12,14,12,14]),'Resumen');
 // Igual que el reporte completo: agendados y adicionales
 // en hojas separadas.
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.filter(c=>grupoReporte(c)==='AGENDA').map(fila)],anchos),
  'Agendados');
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.filter(c=>grupoReporte(c)==='SOBRECUPO').map(fila)],anchos),
  'Adicionales sobrecupo');
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.filter(c=>c.asistencia==='ASISTIO').map(fila)],anchos),
  'Asistieron');
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.filter(c=>c.asistencia==='NO_ASISTIO').map(fila)],anchos),
  'No asistieron');
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.filter(c=>!c.asistencia).map(fila)],anchos),
  'Sin marcar');
 XLSX.utils.book_append_sheet(wb,
  hoja([cab,...d.consultas.map(fila)],anchos),'Detalle completo');

 try {
  XLSX.writeFile(wb,'Reporte_Asistencia_'+d.fecha+'.xlsx');
  MOV.exito();
  aviso('Reporte descargado.');
 }catch(e) {
  aviso('No se pudo descargar: '+e.message);
 }

async function guardarLimites(fila,alcance,restablecer) {
 const esp=fila.dataset.e;
 const cambios=[];

 if(alcance==='FECHA') {
  const input=fila.querySelector('input');
  cambios.push({alcance,fecha:V.fecha,especialidad:esp,
   limite:restablecer?'':input.value.trim()});
 } else {
  fila.querySelectorAll('input[data-dia]').forEach(input=>{
   if(input.value.trim()!==input.dataset.orig) {
    cambios.push({alcance,dia:input.dataset.dia,
     especialidad:esp,limite:input.value.trim()});
   }
  });
 }

 if(!cambios.length) {
  aviso('No hay cambios para guardar.');
  return;
 }

 const r=(V.datos.cupos||[]).find(x=>x.especialidad===esp);
 const nuevo=cambios[0].limite;

 if(alcance==='FECHA'&&r&&nuevo!==''&&Number(nuevo)<r.ocupados&&
    !await confirmar({
     titulo:'LÍMITE MENOR A LO AGENDADO',
     tipo:'urgente',
     html:`<p><b>${esc(esp)}</b> ya tiene <b>${r.ocupados}</b>
      agendados y el nuevo límite es <b>${esc(nuevo)}</b>.</p>
      <p>No se cancela a nadie, pero ya no habrá cupos
      disponibles para esa fecha.</p>`,
     ok:'Guardar de todos modos',
     cancelar:'Volver'
    })) {
  return;
 }

 try {
  for(const c of cambios) {
   await api('actualizarLimite',UGM.token,c);
  }
  if(document.activeElement)document.activeElement.blur();
  fila.querySelectorAll('input.cambiado')
   .forEach(i=>i.classList.remove('cambiado'));
  MOV.exito();
  aviso('Límite actualizado: '+esp+'. Los técnicos lo verán en segundos.');
  V.firmas={};
  await refrescar(true);
 }catch(e) {
  if(sesionTerminada(e,limpiarTodo))return;
  aviso(e.message);
 }
}

function montarLimites() {
 ['cuposRows','generalesRows'].forEach(id=>{
  const cont=$(id);

  cont.addEventListener('click',ev=>{
   const paso=ev.target.closest('[data-paso]');

   if(paso) {
    const input=paso.parentElement.querySelector('input');
    const v=Math.max(0,Math.min(200,
     (Number(input.value)||0)+Number(paso.dataset.paso)));
    input.value=v;
    input.classList.toggle('cambiado',String(v)!==input.dataset.orig);
    MOV.pulso(input);
    return;
   }

   const fila=ev.target.closest('tr[data-e]');
   if(!fila)return;

   if(ev.target.closest('[data-guardar]')) {
    guardarLimites(fila,ev.target.closest('[data-guardar]').dataset.guardar);
   } else if(ev.target.closest('[data-restablecer]')) {
    guardarLimites(fila,'FECHA',true);
   }
  });

  cont.addEventListener('input',ev=>{
   const input=ev.target.closest('input');
   if(input)input.classList.toggle('cambiado',
    input.value.trim()!==input.dataset.orig);
  });
 });
}

  'OBSERVACIONESUCC','OBSERVACIONUCC','OBSERVACIONES',
  'OBSERVACION'
 ],
 especialidad:[
  'ESPECIALIDAD','ESPECIALIDADES',
  'SERVICIO','NOMBREESPECIALIDAD'
 ]
};

function encabezadosPacientes(fila) {
 const nombres=fila.map(nombreColumna);
 const uno=lista=>nombres.findIndex(
  c=>lista.includes(c)
 );
 const todos=lista=>nombres
  .map((c,i)=>lista.includes(c)?i:-1)
  .filter(i=>i>=0);

 const completo=uno(ALIAS_PACIENTES.completo);

 const partes=completo>=0
  ?[completo]
  :[
   ...todos(ALIAS_PACIENTES.nombres),
   ...todos(ALIAS_PACIENTES.apellidos)
  ].sort((a,b)=>a-b);

 const documento=uno(ALIAS_PACIENTES.documento);

 if(!partes.length||documento<0)return null;

 return {
  partes,
  documento,
  expediente:uno(ALIAS_PACIENTES.expediente),
  especialidad:uno(ALIAS_PACIENTES.especialidad),
  fecha:uno(ALIAS_PACIENTES.fecha),
  hora:uno(ALIAS_PACIENTES.hora),
  observacion:uno(ALIAS_PACIENTES.observacion),
  correlativo:uno(ALIAS_PACIENTES.correlativo)
 };
}

async function procesarPacientes(origen) {
 const libro=await leerLibro(origen);

 let mejor=null;

 for(const nombre of libro.SheetNames) {
  const matriz=XLSX.utils.sheet_to_json(
   libro.Sheets[nombre],{
    header:1,
    defval:'',
    raw:false,
    blankrows:true
   }
  );

  for(let i=0;i<Math.min(40,matriz.length);i++) {
   const campos=encabezadosPacientes(matriz[i]);

   if(campos&&(!mejor||
      matriz.length-i>mejor.matriz.length-mejor.inicio)) {
    mejor={matriz,inicio:i+1,campos,hoja:nombre};
    break;
   }
  }
 }

 if(!mejor) {
  throw Error(
   'ARCHIVO_EQUIVOCADO|No encontré las columnas de Nombre y DUI. '+
   'Carga aquí la base de pacientes que envía Reportería.'
  );
 }

 if(esReporteCitas(mejor.matriz[mejor.inicio-1])) {
  throw Error(
   'ARCHIVO_EQUIVOCADO|Este archivo es el Reporte de Citas del '+
   'sistema (la agenda del día), no la base de pacientes. '+
   'Cárgalo en "Agenda". Aquí va la base de pacientes que '+
   'envía Reportería.'
  );
 }

 const {matriz,inicio}=mejor;
 let campos=mejor.campos;
 const registros=[];

 // Valores crudos (fechas como Date) de la misma hoja.
 const crudo=XLSX.utils.sheet_to_json(
  libro.Sheets[mejor.hoja],{header:1,defval:'',raw:true,blankrows:true}
 );

 // La base de Reportería trae varias secciones, cada una
 // con su título y sus encabezados repetidos. Desde la
 // sección "AUMENTO DEL 10%" los pacientes se marcan aparte.
 let aumento=false;
 let ultimaEspecialidad='';

 for(let i=inicio;i<matriz.length;i++) {
  const fila=matriz[i]||[];
  const llenas=fila.filter(v=>String(v??'').trim()!=='');

  // Encabezado repetido: si le falta alguna columna (p. ej.
  // "FECHA Y HORA" en Fisiatría) se conserva la anterior.
  const otraCabecera=encabezadosPacientes(fila);
  if(otraCabecera) {
   ['expediente','especialidad','fecha','hora','observacion','correlativo']
    .forEach(k=>{if(otraCabecera[k]<0)otraCabecera[k]=campos[k];});
   campos=otraCabecera;
   ultimaEspecialidad='';
   continue;
  }

  if(llenas.length===1&&
     /AUMENTO/.test(nombreColumna(llenas[0]))) {
   aumento=true;
   continue;
  }

  const nombre=limpiarNombrePaciente(campos.partes
   .map(c=>String(fila[c]??'').trim())
   .filter(Boolean).join(' '));

  if(!nombre||llenas.length<2)continue;

  const cita=campos.fecha>=0
   ?fechaReporteria((crudo[i]||[])[campos.fecha],fila[campos.fecha])
   :{fecha:'',hora:''};

  if(campos.hora>=0&&!cita.hora) {
   cita.hora=horaReporteria(fila[campos.hora]);
  }

  // Si una fila viene sin especialidad, se usa la de la
  // fila anterior de la misma sección.
  const especialidad=(campos.especialidad>=0
   ?String(fila[campos.especialidad]??'').trim():'')||ultimaEspecialidad;
  ultimaEspecialidad=especialidad;

  registros.push({
   nombre,
   documento:String(fila[campos.documento]??''),
   expediente:campos.expediente>=0
    ?String(fila[campos.expediente]??''):'',
   especialidad,
   fecha:cita.fecha,
   hora:cita.hora,
   aumento,
   observacion:campos.observacion>=0
    ?String(fila[campos.observacion]??'').trim():'',
   correlativo:campos.correlativo>=0
    ?String(fila[campos.correlativo]??'').trim():''
  });
 }

 return registros;
}

// Hora "7:00 am", "1:00 pm", "13:30" -> "07:00", "13:00".
function horaReporteria(v) {
 const m=String(v??'').toLowerCase()
  .match(/(\d{1,2})\s*[:.h]\s*(\d{2})\s*([ap])?\.?\s*m?/);
 if(!m)return '';
 let h=Number(m[1]);
 const min=Number(m[2]);
 if(m[3]==='p'&&h<12)h+=12;
 if(m[3]==='a'&&h===12)h=0;
 if(h>23||min>59)return '';
 return String(h).padStart(2,'0')+':'+String(min).padStart(2,'0');
}

// La columna "FECHA Y HORA" viene escrita de muchas formas:
// celda de fecha, "10-10-2026", "10/10/2026",
// "2026-10-10", "11-10-2026 a las 7:00 am", "11/10/026".
// Se interpreta como día/mes/año (El Salvador).
function fechaReporteria(crudo,texto) {
 const dos=n=>String(n).padStart(2,'0');

 if(crudo instanceof Date&&!isNaN(crudo)) {
  const d=new Date(crudo.getTime()+30000);
  const h=d.getHours()||d.getMinutes()
   ?dos(d.getHours())+':'+dos(d.getMinutes()):'';
  return {fecha:d.getFullYear()+'-'+dos(d.getMonth()+1)+'-'+dos(d.getDate()),hora:h};
 }

 if(typeof crudo==='number'&&crudo>30000&&crudo<80000) {
  const ms=Math.round((crudo-25569)*86400000);
  const d=new Date(ms);
  const h=d.getUTCHours()||d.getUTCMinutes()
   ?dos(d.getUTCHours())+':'+dos(d.getUTCMinutes()):'';
  return {fecha:d.getUTCFullYear()+'-'+dos(d.getUTCMonth()+1)+'-'+dos(d.getUTCDate()),hora:h};
 }

 const t=String(texto??crudo??'').trim();
 const hora=horaReporteria(t.replace(/^\S+/,''));
 let a,m,d,x;

 if((x=t.match(/(\d{4})[-\/.](\d{1,2})[-\/.](\d{1,2})/))) {
  [a,m,d]=[x[1],x[2],x[3]];
 } else if((x=t.match(/(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{2,4})/))) {
  [d,m,a]=[x[1],x[2],x[3]];
  a=Number(a);
  if(a<100)a+=2000;
  else if(a<1000)a=2000+(a%100);
 } else {
  return {fecha:'',hora};
 }

 a=Number(a);m=Number(m);d=Number(d);
 if(m<1||m>12||d<1||d>31)return {fecha:'',hora};
 return {fecha:a+'-'+dos(m)+'-'+dos(d),hora};
}

async function importarPacientes(origen) {
 if(!origen||S.cargandoPacientes)return;

 if(!puedeGestionar()) {
  aviso('Solo el supervisor puede cargar la base.');
  return;
 }

 S.cargandoPacientes=true;

 const progreso=$('patientsProgress');
 const barra=$('patientsBar');
 progreso.classList.remove('hidden');
 barra.classList.remove('hidden');
 MOV.barra(barra.firstElementChild,0);
 progreso.textContent='Leyendo base de pacientes...';

 try {
  const registros=await procesarPacientes(origen);

  if(!registros.length) {
   throw Error('El archivo no contiene pacientes.');
  }

  if(registros.length>250000) {
   throw Error(
    'La base supera 250000 pacientes permitidos.'
   );
  }

  const LOTE=4000;
  const partes=Math.ceil(registros.length/LOTE);
  let carga='';
  let r=null;

  for(let i=0;i<partes;i++) {
   progreso.textContent='Cargando base de pacientes: '+
    Math.round(i/partes*100)+'% ('+
    registros.length+' registros)';
   MOV.barra(barra.firstElementChild,
    Math.max(.04,i/partes));

   r=await api(
    'cargarPacientes',
    S.token,
    carga,
    registros.slice(i*LOTE,(i+1)*LOTE),
    i,
    partes,
    origen.name
   );

   if(!r||!r.ok) {
    throw Error(
     'El servidor no confirmó la carga de pacientes.'
    );
   }

   carga=r.carga;
  }

  const porDia=Object.keys(r.fechas||{}).sort().map(f=>
   fechaLarga(f)+': '+r.fechas[f].total+
   (r.fechas[f].aumento?' ('+r.fechas[f].aumento+' aumento 10%)':''));

  progreso.textContent='Base de pacientes cargada: '+
   r.total+' pacientes · DUI inválidos: '+
   r.invalidos+'.'+(porDia.length?' Por día → '+porDia.join(' · ')+'.':'')+
   ' Ya está disponible para los técnicos.';
  MOV.barra(barra.firstElementChild,1);

  aviso(
   'Base de pacientes actualizada: '+r.total+
   ' pacientes. DUI inválidos: '+r.invalidos+'.'
  );

  V.firmas={};
  await refrescar(true);

 }catch(e) {
  progreso.textContent='No se pudo cargar la base.';
  mostrarError(e);

 }finally {
  S.cargandoPacientes=false;
  $('patientsFile').value='';
 }
}

function limpiarArchivo() {
 if(S.previewURL) {
  URL.revokeObjectURL(S.previewURL);
  S.previewURL=null;
 }

 S.archivo=null;
}

function elegirArchivo(archivo) {
 if(!archivo)return;

 const permitidos=[
  'image/png',
  'image/jpeg',
  'image/webp',
  'application/pdf'
 ];

 if(!permitidos.includes(archivo.type)) {
  aviso('Selecciona JPG, PNG, WEBP o PDF.');
  return;
 }

 if(archivo.size===0 ||
    archivo.size>5*1024*1024) {
  aviso('El archivo debe ser menor de 5 MB.');
  return;
 }

 limpiarArchivo();

 S.archivo=archivo;

 const cont=$('filePreview');

 if(!cont)return;

 cont.classList.remove('hidden');
 cont.innerHTML='';

 const titulo=document.createElement('p');
 titulo.textContent=archivo.name;
 cont.appendChild(titulo);

 if(archivo.type.startsWith('image/')) {
  S.previewURL=URL.createObjectURL(archivo);

  const imagen=document.createElement('img');

  imagen.src=S.previewURL;
  imagen.alt='Comprobante seleccionado';

  cont.appendChild(imagen);

 }else {
  const texto=document.createElement('div');

  texto.className='pdf-preview';
  texto.textContent='PDF: '+archivo.name;

  cont.appendChild(texto);
 }

 const quitar=document.createElement('button');

 quitar.type='button';
 quitar.className='btn';
 quitar.textContent='Quitar comprobante';

 quitar.onclick=()=>{
  limpiarArchivo();
  cont.innerHTML='';
  cont.classList.add('hidden');
  $('fileInput').value='';
 };

 cont.appendChild(quitar);
}

function montarComprobante() {
 const zona=$('dropzone');
 const input=$('fileInput');

 zona.addEventListener('click',()=>{
  input.click();
 });

 zona.addEventListener('keydown',ev=>{
  if(ev.key==='Enter'||ev.key===' ') {
   ev.preventDefault();
   input.click();
  }
 });

 input.addEventListener('change',ev=>{
  elegirArchivo(ev.target.files[0]);
 });

 zona.addEventListener('dragover',ev=>{
  ev.preventDefault();
  zona.classList.add('dragging');
 });

 zona.addEventListener('dragleave',()=>{
  zona.classList.remove('dragging');
 });

 zona.addEventListener('drop',ev=>{
  ev.preventDefault();
  zona.classList.remove('dragging');
  elegirArchivo(ev.dataTransfer.files[0]);
 });
}
