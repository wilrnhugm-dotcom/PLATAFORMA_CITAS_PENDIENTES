  if(!window.google ||
     !google.script ||
     !google.script.run) {
   reject(Error(
    'Abre la aplicación desde Google Apps Script.'
   ));

   .withSuccessHandler(resolve)
   .withFailureHandler(e=>reject(
    Error(e&&e.message?e.message:String(e))
   ))[funcion](...args);
 });

function normalizarDUI(v) {
 const digitos=String(v||'')
  .replace(/\.0+$/,'')
  .replace(/\D/g,'');

 const nueve=digitos.length===8
  ?'0'+digitos:digitos;

 return nueve.length===9
  ?nueve.slice(0,8)+'-'+nueve.slice(8)

}

function normTexto(v) {

  .normalize('NFD')
  .replace(/[̀-ͯ]/g,'')
  .toUpperCase()
  .replace(/[^A-Z0-9 ]/g,' ')
  .replace(/\s+/g,' ')
  .trim();
}

function resaltar(texto,consulta) {
 const limpio=esc(texto);
 const palabras=normTexto(consulta).split(' ')
  .filter(x=>x.length>=2);

 if(!palabras.length)return limpio;

 // Resalta sobre el texto sin acentos, conservando
 // el original.
 const base=String(texto).normalize('NFD')
  .replace(/[̀-ͯ]/g,'').toUpperCase();

 if(base.length!==String(texto).length)return limpio;

 const marcas=new Array(base.length).fill(false);

 palabras.forEach(x=>{
  let i=base.indexOf(x);
  while(i>=0) {
   for(let k=i;k<i+x.length;k++)marcas[k]=true;
   i=base.indexOf(x,i+x.length);
  }
 });

 let html='';
 let abierta=false;

 [...String(texto)].forEach((c,i)=>{
  if(marcas[i]&&!abierta){html+='<mark>';abierta=true;}
  if(!marcas[i]&&abierta){html+='</mark>';abierta=false;}
  html+=esc(c);
 });

 return html+(abierta?'</mark>':'');
}

  .toUpperCase();
}

function formatoFechaExcel(valor) {
 if(valor instanceof Date) {
  if(isNaN(valor.getTime()))return '';

  return [
   valor.getFullYear(),
   String(valor.getMonth()+1).padStart(2,'0'),
   String(valor.getDate()).padStart(2,'0')
  ].join('-');
 }

 if(typeof valor==='number' && window.XLSX) {
  const f=XLSX.SSF.parse_date_code(valor);

  if(f) {
   return `${f.y}-${String(f.m).padStart(2,'0')}`+
    `-${String(f.d).padStart(2,'0')}`;
  }
 }

 const texto=String(valor||'').trim();

 let m=texto.match(
  /^(\d{4})-(\d{1,2})-(\d{1,2})/
 );

 if(m) {
  return `${m[1]}-${m[2].padStart(2,'0')}`+
   `-${m[3].padStart(2,'0')}`;
 }

 m=texto.match(
  /^(\d{1,2})\/(\d{1,2})\/(\d{4})/
 );

 if(m) {
  return `${m[3]}-${m[2].padStart(2,'0')}`+
   `-${m[1].padStart(2,'0')}`;
 }

 return '';
}

// "10553-25 - MARIA ORTIZ" -> "MARIA ORTIZ"
function limpiarNombrePaciente(v) {
 return String(v==null?'':v)
  .replace(/^\s*\d[\d\-\/]*\s*-\s*/,'').trim();
}

// El Reporte de Citas del sistema tiene Fecha/Hora,
// Especialidad y Estado Cita.
function esReporteCitas(fila) {
 const n=(fila||[]).map(nombreColumna);
 return n.includes('ESTADOCITA')||
  (n.includes('FECHAHORA')&&n.includes('ESPECIALIDAD'));
}

function detectarEncabezados(fila) {
 const campos={};

 Object.entries(ALIAS_COLUMNAS).forEach(
  ([campo,alternativas])=>{
   const i=fila.findIndex(c=>
    alternativas.includes(nombreColumna(c))
   );

   if(i>=0)campos[campo]=i;
  }
 );

 if(
  campos.fecha===undefined ||
  campos.especialidad===undefined ||
  campos.estado===undefined
 ) {
  return null;
 }

 return campos;
}

async function leerLibro(origen) {
 if(!window.XLSX) {
  throw Error(
   'No pudo cargarse el lector Excel. '+
   'Comprueba tu conexión.'
  );
 }

 if(typeof origen.texto==='string') {
  // raw: conserva el texto tal cual (ceros
  // iniciales y fechas dd/mm/aaaa).
  return XLSX.read(origen.texto,{
   type:'string',
   raw:true
  });
 }

 const buffer=await origen.arrayBuffer();

 return XLSX.read(buffer,{
  type:'array',
  cellDates:true
 });
}

async function procesarExcel(archivo) {
 const libro=await leerLibro(archivo);

 let mejor=null;
 let total=0;

 // Se revisan todas las hojas hasta encontrar
 // encabezados reconocibles.
 for(const nombre of libro.SheetNames) {
  const hoja=libro.Sheets[nombre];

  const matriz=XLSX.utils.sheet_to_json(
   hoja,{
    header:1,
    defval:'',
    raw:true,
    blankrows:false
   }
  );

  for(let i=0;i<Math.min(25,matriz.length);i++) {
   const campos=detectarEncabezados(matriz[i]);

   if(campos) {
    const cantidad=matriz.length-i-1;

    if(!mejor || cantidad>total) {
     mejor={matriz,inicio:i+1,campos};
     total=cantidad;
    }
   }
  }
 }

 if(!mejor) {
  throw Error(
   'ARCHIVO_EQUIVOCADO|Este archivo no es el Reporte de Citas '+
   'del sistema hospitalario (no tiene las columnas Fecha/Hora, '+
   'Especialidad y Estado Cita). Si es la base de pacientes que '+
   'envía Reportería, cárgala en "Base de pacientes".'
  );
 }

 const registros=[];

 const {matriz,inicio,campos}=mejor;

 for(let i=inicio;i<matriz.length;i++) {
  const fila=matriz[i];

  if(!fila || !fila.some(x=>
   x!=='' && x!==null && x!==undefined
  ))continue;

  function valor(campo) {
   const indice=campos[campo];

   return indice===undefined
    ?''
    :fila[indice];
  }

  registros.push({
   id:String(valor('id')??''),
   fecha:formatoFechaExcel(valor('fecha')),
   especialidad:String(
    valor('especialidad')??''
   ),
   estado:String(valor('estado')??''),
   documento:String(
    valor('documento')??''
   ),
   paciente:limpiarNombrePaciente(valor('paciente'))
  });
 }

 return registros;
}

async function archivoBase64(archivo) {
 const buffer=await archivo.arrayBuffer();

 if(buffer.byteLength>5*1024*1024) {
  throw Error('Archivo mayor de 5 MB.');
 }

 const bytes=new Uint8Array(buffer);
 let binario='';

 for(let i=0;i<bytes.length;i+=8192) {
  binario+=String.fromCharCode(
   ...bytes.subarray(i,i+8192)
  );
 }

 return btoa(binario);
}

 // Aparición escalonada a medida que entran en pantalla.
 revelar(elementos) {
  const lista=Array.from(elementos||[]).filter(el=>
   el&&!el.dataset.revelado&&el.offsetParent!==null
  );

  if(!lista.length)return;

  lista.forEach(el=>{el.dataset.revelado='1';});

  if(!MOV.ok)return;

  if(!ST) {
   MOV.escalonar(lista,lista.length);
   return;
  }

  gsap.set(lista,{opacity:0,y:26});

  ScrollTrigger.batch(lista,{
   start:'top 94%',
   once:true,
   onEnter:lote=>gsap.to(lote,{
    opacity:1,y:0,duration:.6,stagger:.07,
    ease:'power3.out',overwrite:true,
    clearProps:'transform,opacity'
   })
  });

  // Seguridad: nada queda invisible si no hubo scroll.
  setTimeout(()=>{
   lista.forEach(el=>{
    if(el.getBoundingClientRect().top<innerHeight&&
       getComputedStyle(el).opacity==='0') {
     gsap.to(el,{opacity:1,y:0,duration:.4,
      clearProps:'transform,opacity'});
    }
   });
  },1800);
 },

 // Recalcula ScrollTrigger cuando cambia el alto de la
 // página (datos nuevos, cambio de sección).

  let alto=0;
  let espera=null;
  new ResizeObserver(()=>{
   const h=document.documentElement.scrollHeight;

 // Red de partículas que se conectan y reaccionan al cursor.
 constelacion() {
  if(!MOV.ok||document.getElementById('fxCanvas'))return;
  const capa=document.querySelector('.bg-orbs');
  if(!capa)return;

  const cv=document.createElement('canvas');
  cv.id='fxCanvas';
  capa.appendChild(cv);
  const ctx=cv.getContext('2d');
  const dpr=Math.min(2,window.devicePixelRatio||1);
  const raton={x:-9999,y:-9999};
  let W=0,H=0,puntos=[];

  const medir=()=>{
   W=innerWidth;H=innerHeight;
   cv.width=W*dpr;cv.height=H*dpr;
   cv.style.width=W+'px';cv.style.height=H+'px';
   ctx.setTransform(dpr,0,0,dpr,0,0);
   const n=Math.round(Math.min(70,Math.max(24,W*H/26000)));
   puntos=Array.from({length:n},()=>({
    x:Math.random()*W,y:Math.random()*H,
    vx:(Math.random()-.5)*.35,vy:(Math.random()-.5)*.35,
    r:Math.random()*1.6+.6
   }));
  };

  medir();
  window.addEventListener('resize',medir);
  window.addEventListener('pointermove',e=>{raton.x=e.clientX;raton.y=e.clientY;},{passive:true});
  document.addEventListener('pointerleave',()=>{raton.x=raton.y=-9999;});

  const color=()=>document.body.classList.contains('light')
   ?'8,127,172':'19,201,245';

  gsap.ticker.add(()=>{
   if(document.hidden)return;
   const c=color();
   ctx.clearRect(0,0,W,H);
   for(const p of puntos) {
    const dx=p.x-raton.x,dy=p.y-raton.y,d2=dx*dx+dy*dy;
    if(d2<14000) {
     const f=(14000-d2)/14000*.6;
     p.vx+=dx/Math.sqrt(d2||1)*f*.08;
     p.vy+=dy/Math.sqrt(d2||1)*f*.08;
    }
    p.vx*=.985;p.vy*=.985;
    if(Math.abs(p.vx)<.05)p.vx+=(Math.random()-.5)*.04;
    if(Math.abs(p.vy)<.05)p.vy+=(Math.random()-.5)*.04;
    p.x+=p.vx;p.y+=p.vy;
    if(p.x<0||p.x>W)p.vx*=-1;
    if(p.y<0||p.y>H)p.vy*=-1;
   }
   for(let i=0;i<puntos.length;i++) {
    const a=puntos[i];
    for(let j=i+1;j<puntos.length;j++) {
     const b=puntos[j];
     const dx=a.x-b.x,dy=a.y-b.y,d=dx*dx+dy*dy;
     if(d<15000) {
      ctx.strokeStyle='rgba('+c+','+(.16*(1-d/15000))+')';
      ctx.lineWidth=.8;
      ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.stroke();
     }
    }
    ctx.fillStyle='rgba('+c+',.55)';
    ctx.beginPath();ctx.arc(a.x,a.y,a.r,0,Math.PI*2);ctx.fill();
   }
  });
 },

// Iniciales: primer nombre y primer apellido.
function iniciales(nombre) {
 const p=String(nombre||'').trim().split(/\s+/)
  .filter(x=>/^[A-Za-zÁÉÍÓÚÜÑáéíóúüñ]/.test(x));
 const segunda=p.length>=4?p[2]:p.length===3?p[1]:p[1];
 return ((p[0]||'?')[0]+((segunda||'')[0]||''))
  .toUpperCase();
}

// "martes, 13 de octubre de 2026"
function fechaLarga(f) {
 const d=new Date(String(f)+'T12:00:00Z');
 if(isNaN(d.getTime()))return String(f||'');
 return d.toLocaleDateString('es-SV',{
  weekday:'long',day:'numeric',month:'long',
  year:'numeric',timeZone:'UTC'
 });
}

function horaCorta(iso) {
 const d=new Date(iso);
 if(isNaN(d.getTime()))return '';
 return d.toLocaleTimeString('es-SV',{
  timeZone:'America/El_Salvador',
  hour:'2-digit',minute:'2-digit'
 });
}

function fechaHora(iso) {
 const d=new Date(iso);
 if(isNaN(d.getTime()))return String(iso||'');
 return d.toLocaleString('es-SV',{
  timeZone:'America/El_Salvador',
  day:'2-digit',month:'2-digit',
  hour:'2-digit',minute:'2-digit'
 });
}

function hace(segundos) {
 if(segundos==null)return '';
 if(segundos<60)return 'hace '+segundos+' s';
 if(segundos<3600)return 'hace '+Math.floor(segundos/60)+' min';
 return 'hace '+Math.floor(segundos/3600)+' h';
}

function claseEstado(estado) {
 const e=normTexto(estado);
 return e==='EN GESTION'?'gestion':
  e==='REGISTRADO'?'registrado':
  e==='RECHAZADO'?'rechazado':
  e==='ANULADO'?'anulado':
  e==='OBSERVADO'?'observado':
  e==='VALIDACION'?'validacion':'pendiente';
}

function textoEstado(estado) {
 const e=normTexto(estado);
 return {
  'PENDIENTE':'Sin verificar',
  'EN GESTION':'En gestión',
  'VALIDACION':'En validación (otra área)',
  'OBSERVADO':'Observado',
  'REGISTRADO':'Agendado',
  'RECHAZADO':'Rechazado',
  'ANULADO':'Anulado'
 }[e]||estado||'Pendiente';
}

function etiquetaEstado(estado) {
 return '<span class="estado '+claseEstado(estado)+'">'+
  esc(textoEstado(estado))+'</span>';
}

// Reduce fotos grandes (máx. 1800 px) para subirlas
// rápido. PDF se envía tal cual. Máximo 5 MB.
async function prepararArchivo(archivo) {
 if(!archivo)return null;

 const permitidos=[
  'image/png','image/jpeg',
  'image/webp','application/pdf'
 ];

 if(!permitidos.includes(archivo.type)) {
  throw Error('Formato no permitido. Usa JPG, PNG, WEBP o PDF.');
 }

 let blob=archivo;
 let mime=archivo.type;

 if(mime.startsWith('image/')&&archivo.size>600*1024&&
    window.createImageBitmap) {
  try {
   const bmp=await createImageBitmap(archivo);
   const k=Math.min(1,1800/Math.max(bmp.width,bmp.height));
   const c=document.createElement('canvas');
   c.width=Math.round(bmp.width*k);
   c.height=Math.round(bmp.height*k);
   c.getContext('2d').drawImage(bmp,0,0,c.width,c.height);
   const reducido=await new Promise(r=>
    c.toBlob(r,'image/jpeg',.85)
   );
   if(reducido&&reducido.size<archivo.size) {
    blob=reducido;
    mime='image/jpeg';
   }
  }catch(e){}
 }

 if(blob.size===0||blob.size>5*1024*1024) {
  throw Error('El archivo debe pesar menos de 5 MB.');
 }

 return {
  nombre:archivo.name,
  mime,
  base64:await archivoBase64(blob)
 };
}

// Con "campo" el diálogo pide un texto y resuelve con él
// (o false si se cancela): {etiqueta, placeholder, minimo}.
function dialogo({titulo,html,ok,cancelar,tipo,icono:icono_,campo}) {
 return new Promise(resolver=>{
  $('modalContent').innerHTML=`
   <div class="aviso-dia ${tipo||''}">
    <div class="aviso-ico">${icono(icono_||'estetoscopio')}</div>
    <h2>${titulo}</h2>
    ${html}
    ${campo?`<div class="field dlg-campo">
     <label for="dlgTexto">${campo.etiqueta||'Motivo'}</label>
     <textarea id="dlgTexto" rows="3" maxlength="300"
      placeholder="${campo.placeholder||''}"></textarea>
     <small class="dlg-cuenta" id="dlgCuenta">0 / ${campo.minimo||1} caracteres mínimos</small>
    </div>`:''}
    <div class="buttons">
     ${cancelar?`<button class="btn" id="dlgNo" type="button">${cancelar}</button>`:''}
     <button class="btn primary" id="dlgSi" type="button">${ok}</button>
    </div>
   </div>`;

  $('modal').classList.remove('hidden');
  MOV.modal();

  if(MOV.ok)gsap.fromTo('.aviso-ico',{scale:.4,rotate:-30},
   {scale:1,rotate:0,duration:.7,ease:'elastic.out(1,.45)'});

  const fin=v=>{
   $('modal').classList.add('hidden');
   $('modalContent').innerHTML='';
   resolver(v);
  };

  if(campo) {
   const minimo=campo.minimo||1;
   const caja=$('dlgTexto');
   const revisar=()=>{
    const n=caja.value.trim().length;
    $('dlgCuenta').textContent=n+' / '+minimo+' caracteres mínimos';
    $('dlgCuenta').classList.toggle('ok',n>=minimo);
    $('dlgSi').disabled=n<minimo;
   };
   caja.addEventListener('input',revisar);
   revisar();
   $('dlgSi').onclick=()=>{
    const v=caja.value.trim();
    if(v.length<minimo) {MOV.pulso(caja);return;}
    fin(v);
   };
   if($('dlgNo'))$('dlgNo').onclick=()=>fin(false);
   caja.focus();
   return;
  }

  $('dlgSi').onclick=()=>fin(true);
  if($('dlgNo'))$('dlgNo').onclick=()=>fin(false);
  $('dlgSi').focus();
 });
}

/* =====================================================
   CONFIRMACIÓN PROPIA (reemplaza al confirm() del
   navegador, que en Apps Script muestra "Una página
   incorporada en ...googleusercontent.com dice").
   Va en su propia capa, así no cierra la ventana que
   esté abierta debajo (por ejemplo, el detalle de una
   solicitud).
   ===================================================== */

function confirmar({titulo,html,ok,cancelar,tipo,icono:icono_}) {
 return new Promise(resolver=>{
  const capa=document.createElement('div');
  capa.className='overlay confirma';
  capa.setAttribute('role','dialog');
  capa.setAttribute('aria-modal','true');
  capa.innerHTML=`
   <div class="modal glass confirma-caja">
    <div class="aviso-dia ${tipo||'bloqueo'}">
     <div class="aviso-ico">${icono(icono_||'alerta')}</div>
     <h2>${titulo}</h2>
     ${html||''}
     <div class="buttons">
      <button class="btn" type="button" data-r="no">${cancelar||'Cancelar'}</button>
      <button class="btn ${tipo==='info'?'primary':'danger'}" type="button" data-r="si">${ok||'Aceptar'}</button>
     </div>
    </div>
   </div>`;

  document.body.appendChild(capa);

  const fin=v=>{
   document.removeEventListener('keydown',tecla,true);
   const quitar=()=>{capa.remove();resolver(v);};
   if(!MOV.ok){quitar();return;}
   gsap.to(capa,{opacity:0,duration:.18,onComplete:quitar});
  };

  const tecla=ev=>{
   if(ev.key==='Escape'){ev.stopPropagation();fin(false);}
  };
  document.addEventListener('keydown',tecla,true);

  capa.addEventListener('click',ev=>{
   if(ev.target===capa){fin(false);return;}
   const b=ev.target.closest('[data-r]');
   if(b)fin(b.dataset.r==='si');
  });

  if(MOV.ok) {
   gsap.fromTo(capa,{opacity:0},{opacity:1,duration:.2});
   gsap.fromTo(capa.querySelector('.confirma-caja'),
    {y:20,scale:.94,opacity:0},
    {y:0,scale:1,opacity:1,duration:.38,ease:'back.out(1.7)'});
   gsap.fromTo(capa.querySelector('.aviso-ico'),{scale:.4,rotate:-30},
    {scale:1,rotate:0,duration:.7,ease:'elastic.out(1,.45)'});
  }

  capa.querySelector('[data-r="no"]').focus();
 });
}

/* =====================================================
   ZONAS DE ARCHIVO (arrastrar, clic o pegar)
   ===================================================== */

function montarZona(zona,input,alElegir) {
 zona.addEventListener('click',ev=>{
  if(ev.target.closest('.quitar'))return;
  input.click();
 });

 zona.addEventListener('keydown',ev=>{
  if(ev.key==='Enter'||ev.key===' ') {
   ev.preventDefault();
   input.click();
  }
 });

 input.addEventListener('change',ev=>{
  alElegir(ev.target.files[0]);
  input.value='';
 });

 zona.addEventListener('dragover',ev=>{
  ev.preventDefault();
  zona.classList.add('dragging');
 });

 zona.addEventListener('dragleave',()=>{
  zona.classList.remove('dragging');
 });

 zona.addEventListener('drop',ev=>{
  ev.preventDefault();
  zona.classList.remove('dragging');
  alElegir(ev.dataTransfer.files[0]);
 });
}

// Ctrl+V de un archivo o de celdas copiadas de Excel.
function pegarComoTabla(ev,importar) {
 const t=ev.target;

 if(t&&['INPUT','TEXTAREA','SELECT']
   .includes(t.tagName))return false;

 const cd=ev.clipboardData;

 if(!cd)return false;

 const archivo=Array.from(cd.files||[]).find(f=>
  /\.(xlsx|xls|csv|txt)$/i.test(f.name||'')
 );

 if(archivo) {
  ev.preventDefault();
  importar(archivo);
  return true;
 }

 const texto=cd.getData('text/plain')||'';

 if(/[\t;,]/.test(texto)&&/\n/.test(texto.trim())) {
  ev.preventDefault();
  importar({name:'Pegado desde portapapeles',texto});
  return true;
 }

 return false;
}

function imagenPegada(ev) {
 const items=Array.from(
  (ev.clipboardData&&ev.clipboardData.items)||[]
 );

 const item=items.find(x=>
  x.kind==='file'&&x.type.startsWith('image/')
 );

 if(!item)return null;

 const archivo=item.getAsFile();

 if(!archivo)return null;

 const ext=archivo.type==='image/jpeg'?'jpg':
  archivo.type==='image/webp'?'webp':'png';

 return new File([archivo],
  'captura_'+Date.now()+'.'+ext,{type:archivo.type});
}

/* =====================================================
   AGENDA EXCEL (compartida)
   ===================================================== */

async function importarAgendaArchivo(archivo,alTerminar) {
 if(!archivo||UGM.importando)return;

 UGM.importando=true;
 aviso('Procesando agenda Excel...');

 try {
  const registros=await procesarExcel(archivo);

  if(!registros.length) {
   throw Error('El archivo no contiene citas.');
  }

  const r=await api(
   'importarAgenda',UGM.token,registros,archivo.name
  );

  if(!r||!r.ok) {
   throw Error('El servidor no confirmó la importación.');
  }

  MOV.exito();
  aviso('Agenda actualizada: '+r.total+
   ' citas válidas. Omitidas: '+r.omitidas+'.');

  if(alTerminar)await alTerminar(r);

 }catch(e) {
  mostrarError(e);

 }finally {
  UGM.importando=false;
 }
}

function salirLocal(mensaje,alSalir) {
 UGM.token='';
 UGM.nombre='';
 UGM.perfil='';
 UGM.sesion='';
 UGM.generacion++;

 if(alSalir)alSalir();

 $('loginForm').reset();
 $('who').textContent='';
 $('app').classList.add('hidden');
 $('login').classList.remove('hidden');
 $('modal').classList.add('hidden');
 $('modalContent').innerHTML='';
 cerrarError();
 MOV.login();

 if(mensaje)aviso(mensaje);
}

// Si supervisión cerró la sesión o venció, se vuelve
// al acceso.
function sesionTerminada(error,alSalir) {
 const m=String(error&&error.message||error);

 if(/Sesi[oó]n (cerrada|vencida|inv[aá]lida|no autorizada)|Inicia sesi[oó]n/
   .test(m)&&UGM.token) {
  salirLocal('Tu sesión terminó: '+m,alSalir);
  return true;
 }

 return false;
}

function montarAcceso(vista,alEntrar,alSalir) {
 $('loginForm').addEventListener('submit',async ev=>{
  ev.preventDefault();

  const b=$('btnLogin');
  b.disabled=true;
  b.textContent='Ingresando...';

  try {
   const r=await api(
    'entrar',
    $('loginName').value,
    $('loginPass').value,
    vista
   );

   UGM.token=r.token;
   UGM.nombre=r.nombre;
   UGM.perfil=r.perfil;
   UGM.sesion=r.sesion||'';
   UGM.cerrando=false;
   UGM.generacion++;

   await MOV.salirLogin();

   $('login').classList.add('hidden');
   $('app').classList.remove('hidden');
   MOV.restaurarLogin();

   $('loginPass').value='';
   $('who').textContent=
    UGM.nombre+' · '+(ROLES[UGM.perfil]||UGM.perfil);

   await alEntrar(r);

  }catch(e) {
   aviso(e.message);

  }finally {
   b.disabled=false;
   b.textContent='Ingresar al sistema';
  }
 });

 $('btnSalir').addEventListener('click',async ev=>{
  ev.preventDefault();

  if(UGM.cerrando)return;

  UGM.cerrando=true;

  const token=UGM.token;

  salirLocal('',alSalir);

  try {
   if(token)await api('salir',token);
   aviso('Sesión cerrada.');
  }catch(e) {
   aviso('Se volvió al acceso, pero no se pudo '+
    'confirmar el cierre remoto.');
  }finally {
   UGM.cerrando=false;
  }
 });
}

function mostrarPagina(n,titulo,sinAnimar) {
 MOV.reiniciarRevelado($('app'));

 document.querySelectorAll('.page').forEach(el=>{
  el.classList.toggle('hidden',el.id!==n);
 });

 if(!sinAnimar)window.scrollTo({top:0,behavior:'smooth'});

 let activa=null;

 document.querySelectorAll('.nav').forEach(el=>{
  const on=el.dataset.page===n;
  el.classList.toggle('active',on);
  if(on)activa=el;
 });

 MOV.tinta(activa);

 $('title').textContent=titulo;
 $('crumbPage').textContent=titulo;

 if(!sinAnimar) {
  MOV.pagina($(n));
  MOV.titulo($('title'));
  MOV.revelar($(n).querySelectorAll(
   '.kpi, .panel, .card, .cola, .notice'
  ));
 }

 MOV.refrescarScroll();
}
