  timeZone:'America/El_Salvador',
  year:'numeric',
  month:'2-digit',
  day:'2-digit'
 }).formatToParts(new Date());

function diaSemana(fecha) {

  return '';
 }

 ).getUTCDay();

 return n===6?'SABADO':n===0?'DOMINGO':'SEMANA';
}

function formatearDUI(v) {
 const digitos=String(v||'')
  .replace(/\D/g,'').slice(0,9);

 return digitos.length>8
  ?digitos.slice(0,8)+'-'+digitos.slice(8)
  :digitos;
}

// DUI recuperado de la base: ocho dígitos se completan
// con un cero inicial; se exige exactamente nueve.
function normalizarDUI(v) {
 const digitos=String(v||'')
  .replace(/\.0+$/,'')
  .replace(/\D/g,'');

 const nueve=digitos.length===8
  ?'0'+digitos:digitos;

 return nueve.length===9
  ?nueve.slice(0,8)+'-'+nueve.slice(8)
  :'';
}

function revisarDUI() {

    $('doc').value===S.paciente.documento) {
  return;
 }

 const n=$('doc').value.replace(/\D/g,'').length;

 if(n===0) {
  ayuda('docHelper',
   'El guion se agrega automáticamente.');
 } else if(n<9) {
  ayuda('docHelper',
   'DUI incompleto: faltan '+(9-n)+
   ' dígito(s). Debe tener nueve.','bad');
 } else {
  ayuda('docHelper','DUI con nueve dígitos.','ok');
 }
}

function dibujarResultados(r) {
 B.resultados=(r&&r.resultados)||[];
 B.indice=B.resultados.length?0:-1;

 if(!B.resultados.length) {
  mostrarResultados(`<div class="search-empty">${
   r&&r.sinBase
    ?'La base de pacientes aún no está cargada.'
    :'Sin coincidencias en la base de pacientes.'
  }</div>`);
  return;
 }

 mostrarResultados(B.resultados.map((p,i)=>`
  <div class="search-item${i===B.indice?' active':''}"
   role="option" data-i="${i}">
   <b>${esc(p.nombre)}</b>
   <small class="${p.valido?'':'bad'}">${
    p.valido
     ?'DUI '+esc(p.documento)
     :'DUI inválido en la base: '+
      esc(p.original||'vacío')
   }${p.expediente
     ?' · Exp. '+esc(p.expediente):''}</small>

 `).join(''));

 document.querySelectorAll(
  '#patientResults .search-item'
 ).forEach(el=>{
  // mousedown evita que el blur cierre la lista
  el.addEventListener('mousedown',ev=>{
   ev.preventDefault();
   elegirPaciente(Number(el.dataset.i));

function marcarResultado() {

  '#patientResults .search-item'

  if(i===B.indice)el.scrollIntoView({block:'nearest'});
 });

async function buscarPaciente(texto) {
 const seq=++B.seq;

 mostrarResultados(
  '<div class="search-empty">Buscando...</div>'
 );

 try {
  const r=await api(
   'buscarPacientes',S.token,texto
  );

 }catch(e) {

   `<div class="search-empty">${esc(e.message)}</div>`

}

function elegirPaciente(i) {
 const p=B.resultados[i];

function montarBuscador() {
 const input=$('patientName');

 input.addEventListener('input',()=>{
  // Si se edita el nombre elegido, el DUI deja de
  // corresponder y se limpia.
  if(S.paciente&&input.value!==S.paciente.nombre) {
   if($('doc').value===S.paciente.documento) {
    $('doc').value='';
   }
   S.paciente=null;
   ayuda('patientHelper',
    'Escribe al menos 3 letras para buscar en la '+
    'base de pacientes.');
   revisarDUI();
  }

  const texto=input.value.trim();
  const digitos=texto.replace(/\D/g,'');
  const porDui=digitos.length>=4&&
   /^[0-9\s-]+$/.test(texto);

  clearTimeout(B.timer);

  if(!porDui&&texto.replace(/\s/g,'').length<3) {
   cerrarResultados();
   return;
  }

  B.timer=setTimeout(()=>buscarPaciente(texto),350);
 });

 input.addEventListener('keydown',ev=>{
  const abierta=
   !$('patientResults').classList.contains('hidden');

  if(!abierta||!B.resultados.length) {
   if(ev.key==='Escape')cerrarResultados();
   return;
  }

  if(ev.key==='ArrowDown') {
   ev.preventDefault();
   B.indice=(B.indice+1)%B.resultados.length;
   marcarResultado();
  } else if(ev.key==='ArrowUp') {
   ev.preventDefault();
   B.indice=(B.indice-1+B.resultados.length)%
    B.resultados.length;
   marcarResultado();
  } else if(ev.key==='Enter') {
   ev.preventDefault();
   elegirPaciente(B.indice);
  } else if(ev.key==='Escape') {
   cerrarResultados();
  }
 });

 input.addEventListener('blur',()=>{
  setTimeout(cerrarResultados,150);
 });
}

function nombreColumna(v) {
 return String(v==null?'':v)
  .trim().normalize('NFD')
  .replace(/[\u0300-\u036f]/g,'')
  .replace(/[^A-Za-z0-9]/g,'')
  .toUpperCase();
}

const ALIAS_COLUMNAS={
 id:[
  'ID','CODIGO','CODIGOCITA',
  'IDCITA','NUMEROCITA','CORRELATIVO'
 ],
 fecha:[
  'FECHA','FECHACITA',
  'FECHAATENCION','FECHAHORA',
  'FECHAYHORA'
 ],
 especialidad:[
  'ESPECIALIDAD','ESPECIALIDADES',
  'SERVICIO','NOMBREESPECIALIDAD'
 ],
 estado:[
  'ESTADO','ESTADOCITA',
  'ESTADODECITA'
 ],
 documento:[
  'EXPEDIENTE','DOCUMENTO',
  'DUI','NUMEROEXPEDIENTE',
  'NOEXPEDIENTE'
 ],
 paciente:[
  'PACIENTE','NOMBREPACIENTE',
  'NOMBRE','NOMBRES',
  'NOMBRECOMPLETO'
 ]
};

function formatoFechaExcel(valor) {
 if(valor instanceof Date) {
  if(isNaN(valor.getTime()))return '';

  return [
   valor.getFullYear(),
   String(valor.getMonth()+1).padStart(2,'0'),
   String(valor.getDate()).padStart(2,'0')
  ].join('-');
 }

 if(typeof valor==='number' && window.XLSX) {
  const f=XLSX.SSF.parse_date_code(valor);

  if(f) {
   return `${f.y}-${String(f.m).padStart(2,'0')}`+
    `-${String(f.d).padStart(2,'0')}`;
  }
 }

 const texto=String(valor||'').trim();

 let m=texto.match(
  /^(\d{4})-(\d{1,2})-(\d{1,2})/
 );

 if(m) {
  return `${m[1]}-${m[2].padStart(2,'0')}`+
   `-${m[3].padStart(2,'0')}`;
 }

 m=texto.match(
  /^(\d{1,2})\/(\d{1,2})\/(\d{4})/
 );

 if(m) {
  return `${m[3]}-${m[2].padStart(2,'0')}`+
   `-${m[1].padStart(2,'0')}`;
 }

 return '';
}

function detectarEncabezados(fila) {
 const campos={};

 Object.entries(ALIAS_COLUMNAS).forEach(
  ([campo,alternativas])=>{
   const i=fila.findIndex(c=>
    alternativas.includes(nombreColumna(c))
   );

   if(i>=0)campos[campo]=i;
  }
 );

 if(
  campos.fecha===undefined ||
  campos.especialidad===undefined ||
  campos.estado===undefined
 ) {
  return null;
 }

 return campos;
}

// origen: File, o {name,texto} para celdas pegadas.
async function leerLibro(origen) {
 if(!window.XLSX) {
  throw Error(
   'No pudo cargarse el lector Excel. '+
   'Comprueba tu conexión.'
  );
 }

 if(typeof origen.texto==='string') {
  // raw: conserva el texto tal cual (ceros
  // iniciales y fechas dd/mm/aaaa).
  return XLSX.read(origen.texto,{
   type:'string',
   raw:true
  });
 }

 const buffer=await origen.arrayBuffer();

 return XLSX.read(buffer,{
  type:'array',
  cellDates:true
 });
}

async function procesarExcel(archivo) {
 const libro=await leerLibro(archivo);

 let mejor=null;
 let total=0;

 // Se revisan todas las hojas hasta encontrar
 // encabezados reconocibles.
 for(const nombre of libro.SheetNames) {
  const hoja=libro.Sheets[nombre];

  const matriz=XLSX.utils.sheet_to_json(
   hoja,{
    header:1,
    defval:'',
    raw:true,
    blankrows:false
   }
  );

  for(let i=0;i<Math.min(25,matriz.length);i++) {
   const campos=detectarEncabezados(matriz[i]);

   if(campos) {
    const cantidad=matriz.length-i-1;

    if(!mejor || cantidad>total) {
     mejor={matriz,inicio:i+1,campos};
     total=cantidad;
    }
   }
  }
 }

 if(!mejor) {
  throw Error(
   'No encontré las columnas de Fecha, '+
   'Especialidad y Estado en el archivo.'
  );
 }

 const registros=[];

 const {matriz,inicio,campos}=mejor;

 for(let i=inicio;i<matriz.length;i++) {
  const fila=matriz[i];

  if(!fila || !fila.some(x=>
   x!=='' && x!==null && x!==undefined
  ))continue;

  function valor(campo) {
   const indice=campos[campo];

   return indice===undefined
    ?''
    :fila[indice];
  }

  registros.push({
   id:String(valor('id')??''),
   fecha:formatoFechaExcel(valor('fecha')),
   especialidad:String(
    valor('especialidad')??''
   ),
   estado:String(valor('estado')??''),
   documento:String(
    valor('documento')??''
   ),
   paciente:String(valor('paciente')??'')
  });
 }

 return registros;
}

async function importarArchivo(archivo) {
 if(!archivo || S.importando)return;

 S.importando=true;

 $('status').textContent=
  'Procesando agenda Excel...';

 try {
  const registros=await procesarExcel(archivo);

  if(!registros.length) {
   throw Error('El archivo no contiene citas.');
  }

  const r=await api(
   'importarAgenda',
   S.token,
   registros,
   archivo.name
  );

  if(!r||!r.ok) {
   throw Error(
    'El servidor no confirmó la importación.'
   );
  }

  aviso(
   'Agenda actualizada: '+r.total+
   ' citas válidas. Omitidas: '+r.omitidas+'.'
  );

  await refrescar();

  if(r.fechas&&r.fechas.length) {
   const actual=S.fecha;

   if(!r.fechas.includes(actual)) {
    const proxima=r.fechas.find(f=>f>=hoy());
    S.fecha=proxima||
     r.fechas[r.fechas.length-1];

    $('filterDate').value=S.fecha;

    await refrescar();
   }
  }

 }catch(e) {
  mostrarError(e);

 }finally {
  S.importando=false;
  $('excelFile').value='';
 }
}

const ALIAS_PACIENTES={

 ],

  'EXPEDIENTE','NUMEROEXPEDIENTE',
  'NOEXPEDIENTE','NUMERODEEXPEDIENTE'
 ]
};

function encabezadosPacientes(fila) {
 const nombres=fila.map(nombreColumna);
 const uno=lista=>nombres.findIndex(
  c=>lista.includes(c)
 );
 const todos=lista=>nombres
  .map((c,i)=>lista.includes(c)?i:-1)
  .filter(i=>i>=0);

 const completo=uno(ALIAS_PACIENTES.completo);

 const partes=completo>=0
  ?[completo]
  :[
   ...todos(ALIAS_PACIENTES.nombres),
   ...todos(ALIAS_PACIENTES.apellidos)
  ].sort((a,b)=>a-b);

 const documento=uno(ALIAS_PACIENTES.documento);

 if(!partes.length||documento<0)return null;

 return {
  partes,
  documento,
  expediente:uno(ALIAS_PACIENTES.expediente)
 };
}

async function procesarPacientes(origen) {
 const libro=await leerLibro(origen);

 let mejor=null;

 for(const nombre of libro.SheetNames) {
  const matriz=XLSX.utils.sheet_to_json(
   libro.Sheets[nombre],{
    header:1,
    defval:'',
    raw:false,
    blankrows:false
   }
  );

  for(let i=0;i<Math.min(25,matriz.length);i++) {
   const campos=encabezadosPacientes(matriz[i]);

   if(campos&&(!mejor||
      matriz.length-i>mejor.matriz.length-mejor.inicio)) {
    mejor={matriz,inicio:i+1,campos};
    break;
   }
  }
 }

 if(!mejor) {
  throw Error(
   'No encontré las columnas de Nombre y DUI '+
   'en la base de pacientes.'
  );
 }

 const {matriz,inicio,campos}=mejor;
 const registros=[];

 for(let i=inicio;i<matriz.length;i++) {
  const fila=matriz[i]||[];

  const nombre=campos.partes
   .map(c=>String(fila[c]??'').trim())
   .filter(Boolean).join(' ');

  if(!nombre)continue;

  registros.push({
   nombre,
   documento:String(fila[campos.documento]??''),
   expediente:campos.expediente>=0
    ?String(fila[campos.expediente]??''):''
  });
 }

 return registros;
}

async function importarPacientes(origen) {
 if(!origen||S.cargandoPacientes)return;

 if(!puedeGestionar()) {
  aviso('No tienes permiso para cargar la base.');
  return;
 }

 S.cargandoPacientes=true;

 const progreso=$('patientsProgress');
 progreso.classList.remove('hidden');
 progreso.textContent='Leyendo base de pacientes...';

 try {
  const registros=await procesarPacientes(origen);

  if(!registros.length) {
   throw Error('El archivo no contiene pacientes.');
  }

  if(registros.length>250000) {
   throw Error(
    'La base supera 250000 pacientes permitidos.'
   );
  }

  const LOTE=4000;
  const partes=Math.ceil(registros.length/LOTE);
  let carga='';
  let r=null;

  for(let i=0;i<partes;i++) {
   progreso.textContent='Cargando base de pacientes: '+
    Math.round(i/partes*100)+'% ('+
    registros.length+' registros)';

   r=await api(
    'cargarPacientes',
    S.token,
    carga,
    registros.slice(i*LOTE,(i+1)*LOTE),
    i,
    partes,
    origen.name
   );

   if(!r||!r.ok) {
    throw Error(
     'El servidor no confirmó la carga de pacientes.'
    );
   }

   carga=r.carga;
  }

  progreso.textContent='Base de pacientes cargada: '+
   r.total+' pacientes · DUI inválidos: '+
   r.invalidos+'.';

  aviso(
   'Base de pacientes actualizada: '+r.total+
   ' pacientes. DUI inválidos: '+r.invalidos+'.'
  );

  await refrescar();

 }catch(e) {
  progreso.textContent='No se pudo cargar la base.';
  mostrarError(e);

 }finally {
  S.cargandoPacientes=false;
  $('patientsFile').value='';
 }
}

function pegarTabla(ev) {
 if(S.pagina!=='actualizacion')return;

 const t=ev.target;

 if(t&&['INPUT','TEXTAREA','SELECT']
   .includes(t.tagName))return;

 const destino=S.zonaPegado==='pacientes'&&
  puedeGestionar()?'pacientes':'agenda';

 const importar=destino==='pacientes'
  ?importarPacientes:importarArchivo;

 const cd=ev.clipboardData;

 if(!cd)return;

 const archivo=Array.from(cd.files||[]).find(f=>
  /\.(xlsx|xls|csv|txt)$/i.test(f.name||'')
 );

 if(archivo) {
  ev.preventDefault();
  importar(archivo);
  return;
 }

 const texto=cd.getData('text/plain')||'';

 if(/[\t;,]/.test(texto)&&/\n/.test(texto.trim())) {
  ev.preventDefault();
  importar({
   name:'Pegado desde portapapeles',
   texto
  });
 }
}

function elegirArchivo(archivo) {

 const permitidos=[
  'image/png',
  'image/jpeg',
  'image/webp',
  'application/pdf'
 ];

  aviso('Selecciona JPG, PNG, WEBP o PDF.');
  return;
 }

 if(archivo.size===0 ||
    archivo.size>5*1024*1024) {
  aviso('El archivo debe ser menor de 5 MB.');
  return;

 limpiarArchivo();

 S.archivo=archivo;

 const cont=$('filePreview');

 if(!cont)return;

 cont.classList.remove('hidden');
 cont.innerHTML='';

 const titulo=document.createElement('p');
 titulo.textContent=archivo.name;
 cont.appendChild(titulo);

 if(archivo.type.startsWith('image/')) {
  S.previewURL=URL.createObjectURL(archivo);

  const imagen=document.createElement('img');

  imagen.src=S.previewURL;
  imagen.alt='Comprobante seleccionado';

  cont.appendChild(imagen);

 }else {
  const texto=document.createElement('div');

  texto.className='pdf-preview';
  texto.textContent='PDF: '+archivo.name;

  cont.appendChild(texto);
 }

 const quitar=document.createElement('button');

 quitar.type='button';
 quitar.className='btn';
 quitar.textContent='Quitar comprobante';

 quitar.onclick=()=>{
  limpiarArchivo();
  cont.innerHTML='';
  cont.classList.add('hidden');
  $('fileInput').value='';
 };

 cont.appendChild(quitar);
}

async function archivoBase64(archivo) {
 const buffer=await archivo.arrayBuffer();

 if(buffer.byteLength>5*1024*1024) {
  throw Error('Archivo mayor de 5 MB.');
 }

 const bytes=new Uint8Array(buffer);
 let binario='';

 for(let i=0;i<bytes.length;i+=8192) {
  binario+=String.fromCharCode(
   ...bytes.subarray(i,i+8192)
  );
 }

 return btoa(binario);
}

async function gestionar(id) {
 const boton=$('btnSaveState');

 boton.disabled=true;
 boton.textContent='Guardando gestión...';

 try {
  const estado=$('newState').value;
  const observacion=$('obs').value;

  const solicitud=(S.datos.solicitudes||[])
   .find(r=>r.ID===id);

  if(estado==='REGISTRADO' &&
     !S.archivo &&
     !solicitud.ComprobanteID) {
   throw Error(
    'Adjunta un comprobante antes de confirmar.'
   );
  }

  if(S.archivo) {
   const resultado=await api(
    'adjuntarComprobante',
    S.token,
    id,
    S.archivo.name,
    S.archivo.type,
    await archivoBase64(S.archivo)
   );

   if(!resultado||!resultado.ok) {
    throw Error(
     'No se confirmó la carga del comprobante.'
    );
   }
  }

  const resultado=await api(
   'cambiarEstado',
   S.token,
   id,
   estado,
   observacion
  );

  if(!resultado||!resultado.ok) {
   throw Error(
    'No se confirmó el cambio de estado.'
   );
  }

  cerrarModal();
  await refrescar();

  aviso('Gestión guardada correctamente.');

 }catch(e) {
  aviso(e.message);
  boton.disabled=false;
  boton.textContent='Guardar gestión';
 }
}

 }
}
