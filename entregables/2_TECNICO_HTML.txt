<script>
'use strict';

/* =====================================================
   UGM 2026 · PLATAFORMA DE TÉCNICOS
   ===================================================== */

const T={
 pagina:'consultas',
 fecha:'',
 datos:null,
 tab:'',
 filtro:'todos',
 consultando:false,
 timer:null,
 guardando:false,
 firma:'',
 tickets:{frente:null,reverso:null},
 urls:{frente:null,reverso:null},
 cupos:{},
 cuposTimer:null
};

// Compatibilidad con el buscador de pacientes.
const S={
 get token(){return UGM.token;},
 get generacion(){return UGM.generacion;},
 get pagina(){return T.pagina;},
 datos:{},
 paciente:null,
 qResultados:[],
 guardando:false
};

const TITULOS={
 consultas:['Consultas del día','Asistencia de pacientes por especialidad'],
 solicitud:['Solicitud de sobrecupo','Sábado y domingo · solicitudes médicas con ticket'],
 missolicitudes:['Mis solicitudes','Seguimiento de tus solicitudes'],
 buscar:['Buscar paciente','Base de pacientes'],
 actualizacion:['Actualizar agenda','Agenda hospitalaria compartida']
};

function page(n) {
 if(!TITULOS[n])return;

 const cambio=T.pagina!==n;
 T.pagina=n;

 mostrarPagina(n,TITULOS[n][0],!cambio);
 $('subtitle').textContent=TITULOS[n][1];

 if(n==='buscar') {
  pintarBusqueda();
  setTimeout(()=>$('qPaciente').focus(),60);
 }

 if(n==='consultas') {
  requestAnimationFrame(()=>marcarPestana(true));
 }

 if(n==='solicitud')actualizarCupos();
}

function actividadActual() {
 if(T.pagina==='consultas') {
  return 'Consultas · '+(T.tab||'Todas las especialidades');
 }
 if(T.pagina==='solicitud') {
  const p=$('patientName').value.trim();
  return 'Solicitud de sobrecupo'+(p?' · '+p:'');
 }
 return TITULOS[T.pagina][0];
}

/* =====================================================
   SINCRONIZACIÓN
   ===================================================== */

async function refrescar() {
 if(!UGM.token||T.consultando||UGM.importando)return;

 T.consultando=true;

 const gen=UGM.generacion;
 const fecha=T.fecha;

 try {
  const d=await api(
   'tableroTecnico',UGM.token,fecha,actividadActual()
  );

  if(gen!==UGM.generacion||fecha!==T.fecha)return;

  T.datos=d;
  S.datos=d;

  dibujar();
  revisarBase();

  $('sync').textContent='● Sincronizado';
  $('sync').style.color='var(--green)';
  $('status').textContent='Última sincronización: '+
   new Date().toLocaleTimeString('es-SV');

 }catch(e) {
  if(sesionTerminada(e,limpiarTodo))return;
  if(gen!==UGM.generacion)return;

  $('sync').textContent='● Error';
  $('sync').style.color='var(--red)';
  $('status').textContent=e.message;

 }finally {
  if(gen===UGM.generacion)T.consultando=false;
 }
}

function dibujar() {
 const d=T.datos;
 if(!d)return;

 pintarIndicadores();
 pintarPestanas();
 pintarConsultas(false);
 pintarMias();

 const a=d.agenda||{};

 $('uploadStatus').textContent=a.fecha
  ?'Última carga: '+a.archivo+' · Responsable: '+a.usuario+
   ' · Antigüedad: '+a.minutos+' minutos.'
  :'Todavía no se ha importado una agenda.';

 $('agendaAlert').classList.toggle('hidden',!a.extraordinaria);

 $('agendaAlertText').textContent=a.extraordinaria
  ?'Supervisión solicitó una actualización extraordinaria '+
   'de la agenda.'
  :'Para sobrecupos de sábado y domingo la agenda debe '+
   'haberse cargado en los últimos 30 minutos.';
}

/* =====================================================
   CONSULTAS POR ESPECIALIDAD Y ASISTENCIA
   ===================================================== */

function consultas() {
 return (T.datos&&T.datos.consultas)||[];
}

function pintarIndicadores() {
 const lista=consultas();
 const si=lista.filter(c=>c.asistencia==='ASISTIO').length;
 const no=lista.filter(c=>c.asistencia==='NO_ASISTIO').length;
 const pend=lista.length-si-no;
 const avance=lista.length?(si+no)/lista.length:0;

 contarNumero('kCitas',lista.length);
 contarNumero('kSi',si);
 contarNumero('kNo',no);
 contarNumero('kPend',pend);

 $('kAvance').textContent=Math.round(avance*100)+'% registrado';
 $('kFecha').textContent=T.fecha===(T.datos||{}).hoy
  ?'Hoy':T.fecha;

 MOV.anillo($('ringAvance'),avance);
}

function pintarPestanas() {
 const grupos={};

 consultas().forEach(c=>{
  const g=grupos[c.especialidad]||
   (grupos[c.especialidad]={total:0,pend:0});
  g.total++;
  if(!c.asistencia)g.pend++;
 });

 const nombres=Object.keys(grupos)
  .sort((a,b)=>grupos[b].total-grupos[a].total||
   a.localeCompare(b));

 if(T.tab&&!grupos[T.tab])T.tab='';

 const firma=nombres.map(n=>n+grupos[n].total+'/'+grupos[n].pend)
  .join('|')+'#'+T.tab;

 if(firma===T.firmaTabs)return;

 const nueva=!T.firmaTabs||
  T.firmaTabs.split('#')[0].replace(/\d+\/\d+/g,'')!==
  firma.split('#')[0].replace(/\d+\/\d+/g,'');

 T.firmaTabs=firma;

 const cont=$('cTabs');
 const ink=cont.querySelector('.tab-ink');

 cont.innerHTML='';
 cont.appendChild(ink);

 const crear=(valor,texto,cuenta,pend)=>{
  const b=document.createElement('button');
  b.type='button';
  b.className='tab'+(valor===T.tab?' active':'');
  b.dataset.tab=valor;
  b.title=pend?pend+' por marcar':'Todo marcado';
  b.innerHTML=esc(texto)+' <b>'+cuenta+'</b>';
  cont.appendChild(b);
 };

 crear('','Todas',consultas().length,
  consultas().filter(c=>!c.asistencia).length);

 nombres.forEach(n=>crear(n,n,grupos[n].total,grupos[n].pend));

 requestAnimationFrame(()=>marcarPestana(true));

 if(nueva)MOV.escalonar(cont.querySelectorAll('.tab'),20);
}

function marcarPestana(inmediato) {
 const cont=$('cTabs');
 const activa=cont.querySelector('.tab.active');
 MOV.pestana(cont,activa,inmediato);
}

function elegirPestana(valor) {
 T.tab=valor;

 $('cTabs').querySelectorAll('.tab').forEach(b=>{
  b.classList.toggle('active',b.dataset.tab===valor);
 });

 T.firmaTabs=(T.firmaTabs||'').split('#')[0]+'#'+valor;

 const activa=$('cTabs').querySelector('.tab.active');
 if(activa)activa.scrollIntoView({block:'nearest',inline:'nearest',behavior:'smooth'});

 marcarPestana(false);
 pintarConsultas(true);
}

function filtrarConsultas() {
 const texto=$('cBuscar').value.trim();
 const q=normTexto(texto);
 const palabras=q.split(' ').filter(Boolean);
 const digitos=texto.replace(/\D/g,'');
 const numerico=/^[\d\s-]+$/.test(texto)&&digitos.length>=3;

 return consultas().filter(c=>{
  if(T.tab&&c.especialidad!==T.tab)return false;

  if(T.filtro==='pendientes'&&c.asistencia)return false;
  if(['ASISTIO','NO_ASISTIO'].includes(T.filtro)&&
     c.asistencia!==T.filtro)return false;

  if(!texto)return true;

  if(numerico) {
   return String(c.documento).replace(/\D/g,'')
    .includes(digitos);
  }

  const b=normTexto(c.paciente+' '+c.documento);
  return palabras.every(p=>b.includes(p));
 });
}

function etiquetaCita(c) {
 if(c.estadoCita==='SOBRECUPO') {
  return '<span class="tag sobrecupo">Sobrecupo · '+
   esc(textoEstado(c.sobrecupo))+'</span>';
 }

 const tag=c.estadoCita==='ACTIVA'
  ?'<span class="tag activa">Cita activa</span>'

async function marcar(clave,valor,boton) {
 const c=consultas().find(x=>x.clave===clave);

 if(!c||c._ocupado)return;

 // Pulsar el estado ya marcado lo quita.
 const nuevo=c.asistencia===valor?'':valor;
 const anterior={
  asistencia:c.asistencia,
  marcadoPor:c.marcadoPor,
  hora:c.hora
 };

 c._ocupado=true;
 c.asistencia=nuevo;
 c.marcadoPor=nuevo?UGM.nombre:'';
 c.hora=nuevo?new Date().toISOString():'';

 actualizarFila(c,boton);

 try {
  const r=await api('marcarAsistencia',UGM.token,{
   fecha:T.fecha,
   especialidad:c.especialidad,
   documento:c.documento,
   paciente:c.paciente,
   origen:c.origen,
   referencia:c.referencia,
   asistencia:nuevo
  });

  c.marcadoPor=r.marcadoPor;
  c.hora=r.hora;

  aviso(nuevo
   ?(nuevo==='ASISTIO'?'Asistencia':'Inasistencia')+
    ' registrada: '+c.paciente
   :'Marca eliminada: '+c.paciente);

 }catch(e) {
  if(sesionTerminada(e,limpiarTodo))return;
  Object.assign(c,anterior);
  aviso('No se guardó: '+e.message);

 }finally {
  c._ocupado=false;
  actualizarFila(c);
 }
}

function actualizarFila(c,boton) {
 const fila=$('cLista').querySelector(
  '.cita[data-k="'+CSS.escape(c.clave)+'"]'
 );

 pintarIndicadores();
 T.firmaTabs='';
 pintarPestanas();

 if(!fila)return;

 // Si el filtro ya no incluye la fila, se redibuja.
 if(T.filtro!=='todos') {
  T.firma='';
  pintarConsultas(false);
  return;
 }

 const tmp=document.createElement('div');
 tmp.innerHTML=htmlCita(c,$('cBuscar').value.trim());
 const nueva=tmp.firstElementChild;
 fila.replaceWith(nueva);

 if(boton) {
  MOV.marca(nueva,nueva.querySelector(
   'button[data-a="'+boton.dataset.a+'"]'
  ));
 }
}

function revisarFecha() {
 const f=$('date').value;
 const bloqueada=fechaBloqueada(f);
 const pasada=f&&T.datos&&f<T.datos.hoy;
 const entreSemana=f&&!esFinDeSemana(f);

 $('date').classList.toggle('blocked',bloqueada||pasada);

 ayuda('dateHelper',
  bloqueada?'No disponible: fecha bloqueada para '+
   'programaciones y cupos.':
  pasada?'No se pueden solicitar fechas pasadas.':
  entreSemana?'Día entre semana: solo con solicitud médica '+
   'y ticket sellado y firmado.':
  f?'Sábado o domingo: sujeto a cupos disponibles.':
  'Fechas no disponibles:',
  bloqueada||pasada?'bad':entreSemana?'':f?'ok':'');

 $('blockedList').innerHTML=FECHAS_BLOQUEADAS.map(x=>{
  const [y,m,d]=x.split('-');
  return `<span class="blocked-chip${x===f?' current':''}"
   title="No disponible">${d}/${m}/${y}</span>`;
 }).join('');

 const mostrarMedico=Boolean(entreSemana&&!bloqueada);
 const wrap=$('medicoWrap');

 if(mostrarMedico&&wrap.classList.contains('hidden')) {
  wrap.classList.remove('hidden');
  MOV.pulso(wrap);
 } else if(!mostrarMedico) {
  wrap.classList.add('hidden');
  $('medico').checked=false;
 }


 T.bloqueoFecha=Boolean(bloqueada||pasada||
  (mostrarMedico&&!$('medico').checked));

 if(!esFinDeSemana(f))T.cupo=null;

 revisarBoton();
 actualizarCupos();
}

async function pintarCupos() {
 const f=$('date').value;
 const esp=$('esp').value;
 const box=$('cupoBox');

 const poner=(num,html,clase)=>{
  const antes=box.className;
  box.className='cupo-box'+(clase?' '+clase:'');
  if(T.cupoNum!==String(num)) {
   T.cupoNum=String(num);
   $('cupoNum').innerHTML=String(num).startsWith('<')?num:esc(num);
   MOV.cifra($('cupoNum'));
   if(clase==='sin'&&antes!==box.className)MOV.sacudir(box);
  }
  $('cupoTexto').innerHTML=html;
 };

 if(!f||!esp) {
  poner('—','Selecciona especialidad y fecha para ver '+
   'los cupos disponibles.');
  return;
 }

 if(fechaBloqueada(f)) {
  poner(icono('bloqueo'),'<b>Fecha no disponible.</b> No se permiten '+
   'nuevas programaciones ni cupos.','sin');
  return;
 }

 if(!esFinDeSemana(f)) {
  poner(icono('estetoscopio'),'<b>Solicitud médica.</b> Fuera de fin de semana '+
   'no se calculan cupos: requiere ticket sellado y '+
   'firmado y la aprobación de supervisión.','medico');
  return;
 }

 const cache=T.cupos[f];

 if(!cache||Date.now()-cache.t>10000) {
  poner('…','Consultando cupos...');

  try {
   const r=await api('consultarCupos',UGM.token,f);
   T.cupos[f]={t:Date.now(),r};
  }catch(e) {
   poner('!','No se pudo consultar: '+esc(e.message),'sin');
   return;
  }

  if($('date').value!==f||$('esp').value!==esp)return;
 }

 const r=T.cupos[f].r;
 const fila=(r.lista||[]).find(x=>
  normTexto(x.especialidad)===normTexto(esp)
 );

 T.cupo=fila||null;

 if(!fila||!fila.habilitada) {
  poner('0','<b>'+esc(esp)+'</b> no tiene sobrecupos '+
   'habilitados ese día.','sin');
  revisarBoton();
  return;
 }

 const clave=f+'|'+esp;

 if(fila.disponibles===0) {
  poner('0','<b>Cupo completo · restricción médica.</b> '+
   esc(esp)+' ya tiene '+fila.ocupados+' de '+fila.limite+
   ' sobrecupos para el '+esc(fechaLarga(f))+
   '. No se pueden agendar más.','sin');
  revisarBoton();
  if(T.avisoCompleto!==clave) {
   T.avisoCompleto=clave;
   avisoCompleto(fila,f);
  }
  return;
 }

 poner(fila.disponibles,
  (fila.ultimo?'<b style="color:var(--yellow)">Último cupo disponible</b>':
   '<b>Cupos disponibles</b>')+' de <b>'+fila.limite+'</b> para '+
  esc(esp)+' · '+fila.ocupados+' agendado(s)'+
  (fila.porFecha?' · límite ajustado por supervisión para esta fecha':''),
  fila.ultimo?'ultimo':'');
 revisarBoton();

}

function pintarTicket(cual) {

 if(T.urls[cual]) {
  URL.revokeObjectURL(T.urls[cual]);
  T.urls[cual]=null;
 }

 if(!archivo) {
  zona.classList.remove('listo');
  zona.innerHTML=cual==='frente'
   ?'<span class="ico">'+icono('ticket')+'</span><strong>Frente del ticket</strong>'+
    '<small>Arrastra, pega (Ctrl+V) o haz clic</small>'+
    '<small>JPG, PNG, WEBP o PDF · máx. 5 MB</small>'
   :'<span class="ico">'+icono('sello')+'</span><strong>Reverso sellado y firmado</strong>'+
    '<small>Sello y firma del médico visibles</small>'+
    '<small>JPG, PNG, WEBP o PDF · máx. 5 MB</small>';
  return;
 }

 zona.classList.add('listo');

 let vista;

 if(archivo.type.startsWith('image/')) {
  T.urls[cual]=URL.createObjectURL(archivo);
  vista='<img alt="Ticket" src="'+T.urls[cual]+'">';
 } else {
  vista='<span class="ico">'+icono('archivo')+'</span><strong>PDF</strong>';
 }

 zona.innerHTML=vista+'<small>'+esc(archivo.name)+'</small>'+

 zona.querySelector('.quitar').onclick=ev=>{
  ev.stopPropagation();
  T.tickets[cual]=null;
  pintarTicket(cual);

 MOV.pulso(zona);
}

function elegirTicket(cual,archivo) {
 if(!archivo)return;

 const permitidos=[
  'image/png','image/jpeg','image/webp','application/pdf'
 ];

 if(!permitidos.includes(archivo.type)) {
  aviso('Selecciona JPG, PNG, WEBP o PDF.');
  return;
 }

 if(archivo.size>15*1024*1024) {
  aviso('El archivo es demasiado grande.');
  return;
 }

 T.tickets[cual]=archivo;
 pintarTicket(cual);
 actualizarProgreso();
}

async function enviar(ev) {
 ev.preventDefault();

 if(S.guardando)return;

 const datos={
  especialidad:$('esp').value,
  nombres:$('patientName').value.trim(),
  documento:$('doc').value.trim(),
  fecha:$('date').value,
  solicitadoPor:$('requestedBy').value.trim(),
  tipo:$('medico').checked?'MEDICO':'FIN_DE_SEMANA'
 };

 if(!/^[0-9]{8}-[0-9]$/.test(datos.documento)) {
  mostrarError('DUI_INVALIDO|El DUI debe contener exactamente '+
   'nueve dígitos en formato 00000000-0.');
  return;
 }

 if(fechaBloqueada(datos.fecha)) {
  mostrarError('FECHA_NO_DISPONIBLE|La fecha '+datos.fecha+
   ' no está disponible.');
  return;
 }

 const entreSemana=!esFinDeSemana(datos.fecha);

 if(entreSemana&&!$('medico').checked) {
  mostrarError('DIA_NO_HABILITADO|Los sobrecupos son solo '+
   'sábado y domingo. Para otro día marca "Solicitud médica".');
  return;
 }

 if(!T.tickets.frente||!T.tickets.reverso) {
  mostrarError('TICKET_REQUERIDO|Adjunta el ticket de la '+
   'consulta: frente y reverso sellado y firmado por el médico.');
  return;
 }

 if(!entreSemana) {
  let fila;

  try {
   fila=await cuposAlDia(datos.fecha,datos.especialidad);
  }catch(e) {
   mostrarError(e);
   return;
  }

  T.cupo=fila;
  T.avisoCompleto=datos.fecha+'|'+datos.especialidad;
  pintarCupos();

  if(!fila||!fila.habilitada||fila.disponibles===0) {
   if(fila&&fila.habilitada)await avisoCompleto(fila,datos.fecha);
   else mostrarError('ESPECIALIDAD_NO_HABILITADA|'+datos.especialidad+
    ' no tiene sobrecupos habilitados ese día.');
   return;
  }

  if(fila.disponibles===1) {
   const ok=await dialogo({
    tipo:'info',
    icono:'info',
    titulo:'ÚLTIMO CUPO DISPONIBLE',
    html:`<p>Vas a agendar el <b>último sobrecupo</b>
     (${fila.limite} de ${fila.limite}) de
     <b>${esc(datos.especialidad)}</b> para el</p>
     <div class="dia-grande">${esc(fechaLarga(datos.fecha))}</div>
     <p>Después de este registro <b>ya no se podrán agendar
     más</b> pacientes ese día por restricción médica.</p>`,
    ok:'Agendar último cupo',
    cancelar:'Revisar'
   });
   if(!ok)return;
  }
 }

 if(entreSemana) {
  const ok=await dialogo({
   titulo:'CONFIRMAR SOBRECUPO ENTRE SEMANA',
   html:`<div class="dia-grande">${esc(fechaLarga(datos.fecha))}</div>
    <p><b>${esc(datos.nombres)}</b> · DUI ${esc(datos.documento)}<br>
    ${esc(datos.especialidad)} · indicado por <b>${esc(datos.solicitadoPor)}</b></p>
    <p>La solicitud quedará <b>pendiente de aprobación</b>
    de supervisión.</p>`,
   ok:'Confirmar y enviar',
   cancelar:'Revisar'
  });
  if(!ok)return;
 }

 const boton=$('btnRequest');

 S.guardando=true;
 boton.disabled=true;
 boton.textContent='Preparando ticket...';

 try {
  const archivos={
   frente:await prepararArchivo(T.tickets.frente),
   reverso:await prepararArchivo(T.tickets.reverso)
  };

  boton.textContent='Enviando solicitud...';

  const r=await api('crearSolicitud',UGM.token,datos,archivos);

  if(!r||!r.ok||!r.id) {
   throw Error('No se confirmó el registro.');
  }

  MOV.exito();
  aviso('Solicitud enviada a supervisión. Referencia: '+r.id);

  if(!r.medico&&r.disponibles===0) {
   setTimeout(()=>dialogo({
    tipo:'info',
    icono:'info',
    titulo:'CUPO COMPLETO',
    html:`<p>Se agendó el último sobrecupo de
     <b>${esc(datos.especialidad)}</b> para el</p>
     <div class="dia-grande">${esc(fechaLarga(datos.fecha))}</div>
     <p>Ya no se podrán agendar más pacientes ese día.</p>`,
    ok:'Entendido'
   }),1100);
  }

  T.cupos={};
  limpiarFormulario();
  await refrescar();

 }catch(e) {
  if(sesionTerminada(e,limpiarTodo))return;
  mostrarError(e);

 }finally {
  S.guardando=false;
  boton.disabled=false;
  boton.innerHTML=icono('enviar')+' Enviar solicitud';
  revisarFecha();
 }
}

// Al escribir el DUI se busca en la base y se completa
// el nombre (y la especialidad si el paciente tiene una).
async function buscarPorDui() {
 const valor=$('doc').value;
 const digitos=valor.replace(/\D/g,'');

 if(S.paciente&&S.paciente.documento===valor) {
  cerrarDocResultados();
  return;
 }

 if(digitos.length<4) {
  cerrarDocResultados();
  return;
 }

 let lista;

 if(P.listo) {
  lista=buscarLocal(digitos,'',8,false).map(vistaPaciente);
  if(digitos.length===8) {
   buscarLocal('0'+digitos,'',8,false).map(vistaPaciente)
    .forEach(p=>{
     if(!lista.some(x=>x.documento===p.documento))lista.push(p);
    });
  }
 } else {
  if(digitos.length<8)return;
  const seq=++D.seq;
  try {
   const r=await api('buscarPacientes',UGM.token,digitos);
   if(seq!==D.seq)return;
   lista=(r&&r.resultados)||[];
  }catch(e) {
   return;
  }
 }

 // Exacto con 9 dígitos, o con 8 aplicando el cero inicial.
 const exactos=lista.filter(p=>{
  const d=String(p.documento).replace(/\D/g,'');
  return d===digitos||(digitos.length===8&&d==='0'+digitos);
 });

 if(digitos.length>=8&&exactos.length===1) {
  cerrarDocResultados();
  aplicarPaciente(exactos[0]);
  return;
 }

 mostrarDocResultados(lista,digitos);
}

function montarDui() {
 const input=$('doc');

 input.addEventListener('keydown',ev=>{
  if($('docResults').classList.contains('hidden'))return;
  if(ev.key==='Enter'&&D.resultados[0]) {
   ev.preventDefault();
   aplicarPaciente(D.resultados[0]);
   cerrarDocResultados();
  } else if(ev.key==='Escape') {
   cerrarDocResultados();
  }
 });

 input.addEventListener('blur',()=>setTimeout(cerrarDocResultados,150));

 $('docResults').addEventListener('mousedown',ev=>{
  const item=ev.target.closest('.search-item');
  if(!item)return;
  ev.preventDefault();
  aplicarPaciente(D.resultados[Number(item.dataset.i)]);
  cerrarDocResultados();
 });
}

const P={
 version:'',lista:[],listo:false,
 cargando:false,progreso:0,error:'',
 agendaVersion:'',agenda:null,
 cargandoAgenda:false
};

function reiniciarBase() {
 Object.assign(P,{
  version:'',lista:[],listo:false,
  cargando:false,progreso:0,error:'',
  agendaVersion:'',agenda:null,
  cargandoAgenda:false
 });
}

function crearRegistro(f) {
 const [nombre,dui,original,expediente,esp]=f;
 const od=String(original||'')
  .replace(/\.0+$/,'').replace(/\D/g,'');

 const sb=String(esp||'').split(' | ')
  .map(x=>x.trim()).filter(Boolean);

 return {
  n:nombre,
  d:dui,
  o:original,
  e:expediente,
  b:normTexto(nombre),
  dd:String(dui||'').replace(/\D/g,''),
  od,
  ex:alfaNum(expediente),
  sb,
  s:sb,
  sn:sb.map(normTexto)
 };
}

function aplicarAgenda() {
 const a=P.agenda;

 P.lista.forEach(r=>{
  if(!a){
   r.s=r.sb;
  } else {
   const set=new Set(r.sb);
   [
    a.porDoc.get(r.d),
    a.porDoc.get(r.ex),
    a.porDoc.get(r.od),
    a.porNombre.get(r.b)
   ].forEach(x=>{
    if(x)x.forEach(e=>set.add(e));
   });
   r.s=[...set];
  }
  r.sn=r.s.map(normTexto);
 });
}

function revisarBase() {
 const info=S.datos.pacientes||{};

 if(info.version&&info.version!==P.version&&
    !P.cargando) {
  descargarBase();
 }

 if(!info.version&&P.version) {
  const agenda=P.agenda;
  const agendaVersion=P.agendaVersion;
  reiniciarBase();
  P.agenda=agenda;
  P.agendaVersion=agendaVersion;
 }

 const ag=(S.datos.agenda||{}).cargaID||'';

 if(ag!==P.agendaVersion&&!P.cargandoAgenda) {
  cargarMapaAgenda(ag);
 }

 pintarEstadoBase();
}

async function descargarBase() {
 P.cargando=true;
 P.progreso=0;
 P.error='';
 pintarEstadoBase();

 const gen=S.generacion;
 const token=S.token;

 try {
  let version='';
  let desde=0;
  let filas=[];
  let reintentos=0;

  for(;;) {
   const r=await api(
    'descargarPacientes',token,version,desde
   );

   if(gen!==S.generacion)return;

   if(r.cambio) {
    if(++reintentos>3)throw Error(
     'La base cambió durante la descarga.'
    );
    version='';
    desde=0;
    filas=[];
    continue;
   }

   version=r.version;

   if(!version)break;

   for(const f of r.filas)filas.push(f);

   desde+=r.filas.length;
   P.progreso=r.total
    ?Math.round(desde/r.total*100):100;
   pintarEstadoBase();

   if(r.fin||!r.filas.length)break;
  }

  P.lista=filas.map(crearRegistro);
  P.version=version;
  P.listo=Boolean(version);
  aplicarAgenda();

  if(S.pagina==='buscar')pintarBusqueda();

 }catch(e) {
  if(gen===S.generacion)P.error=e.message;

 }finally {
  if(gen===S.generacion) {
   P.cargando=false;
   pintarEstadoBase();
  }
 }
}

async function cargarMapaAgenda(version) {
 P.cargandoAgenda=true;

 const gen=S.generacion;

 try {
  const r=await api('mapaAgendaPacientes',S.token);

  if(gen!==S.generacion)return;

  const porDoc=new Map();
  const porNombre=new Map();

  const agregar=(mapa,clave,esp)=>{
   if(!clave)return;
   if(!mapa.has(clave))mapa.set(clave,new Set());
   mapa.get(clave).add(esp);
  };

  (r.filas||[]).forEach(([dui,alfa,nombre,esp])=>{
   agregar(porDoc,dui,esp);
   agregar(porDoc,alfa,esp);
   agregar(porNombre,nombre,esp);
  });

  P.agenda={porDoc,porNombre};
  aplicarAgenda();

  if(S.pagina==='buscar')pintarBusqueda();

 }catch(e) {
  console.error(e);

 }finally {
  if(gen===S.generacion) {
   // Se marca aunque falle, para no reintentar
   // cada 10 segundos.
   P.agendaVersion=version;
   P.cargandoAgenda=false;
  }
 }
}

function buscarLocal(texto,especialidad,limite,estricto) {
 const t=String(texto||'').trim();
 const q=normTexto(t);
 const compacto=t.replace(/[\s-]/g,'');
 const digitos=t.replace(/\D/g,'');
 const numerico=/^\d+$/.test(compacto)&&
  digitos.length>=3;
 const esp=especialidad?normTexto(especialidad):'';
 const palabras=numerico
  ?[]:q.split(' ').filter(Boolean);

 if(!numerico&&q.replace(/ /g,'').length<2&&
    !(esp&&estricto)) {
  return [];
 }

 const encontrados=[];

 for(const r of P.lista) {
  if(esp&&estricto&&!r.sn.includes(esp))continue;

  let ok;

  if(numerico) {
   ok=r.dd.startsWith(digitos)||
    r.od.startsWith(digitos)||
    r.ex.includes(digitos);
  } else if(palabras.length) {
   ok=palabras.every(x=>r.b.includes(x))||
    (palabras.length===1&&
     r.ex.includes(palabras[0]));
  } else {
   ok=true;
  }

  if(!ok)continue;

  encontrados.push(r);

  if(encontrados.length>=2000)break;
 }

 const primera=palabras[0]||'';

 const puntaje=r=>
  (primera&&!r.b.startsWith(primera)?2:0)+
  (esp&&!estricto&&!r.sn.includes(esp)?1:0);

 return encontrados
  .map(r=>({r,p:puntaje(r)}))
  .sort((a,b)=>a.p-b.p)
  .slice(0,limite)
  .map(x=>x.r);
}

}

   revisarDUI();

  const digitos=texto.replace(/\D/g,'');

   ()=>buscarPaciente(texto),

  );
 });

 $('patientResults').addEventListener('mousedown',ev=>{
  const item=ev.target.closest('.search-item');
  if(!item)return;
  ev.preventDefault();
  elegirPaciente(Number(item.dataset.i));
 });

 input.addEventListener('keydown',ev=>{
  const abierta=
   !$('patientResults').classList.contains('hidden');

  if(!abierta||!B.resultados.length) {
   if(ev.key==='Escape')cerrarResultados();
   return;
  }

  if(ev.key==='ArrowDown') {
   ev.preventDefault();
   B.indice=(B.indice+1)%B.resultados.length;
   marcarResultado();
  } else if(ev.key==='ArrowUp') {
   ev.preventDefault();
   B.indice=(B.indice-1+B.resultados.length)%
    B.resultados.length;
   marcarResultado();
  } else if(ev.key==='Enter') {
   ev.preventDefault();
   elegirPaciente(B.indice);
  } else if(ev.key==='Escape') {
   cerrarResultados();
  }
 });

 input.addEventListener('blur',()=>{
  setTimeout(cerrarResultados,150);
 });

    $('doc').value===S.paciente.documento) {

 if(n===0) {

   'El guion se agrega automáticamente.');
 } else if(n<9) {
  ayuda('docHelper',
   'DUI incompleto: faltan '+(9-n)+
   ' dígito(s). Debe tener nueve.','bad');
 } else {
  ayuda('docHelper','DUI con nueve dígitos.','ok');
 }
}
